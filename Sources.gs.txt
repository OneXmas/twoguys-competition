function JSONLD_MENU(pageUrl) {
  const html = UrlFetchApp.fetch(pageUrl, { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0' } }).getContentText();
  const rows = [['Item', 'Price', 'Section']];
  const re = /<script type="application\/ld\+json">\s*([\s\S]*?)\s*<\/script>/g;
  let m;
  while ((m = re.exec(html)) !== null) {
    let d; try { d = JSON.parse(m[1]); } catch (e) { continue; }
    if (d['@type'] !== 'Menu') continue;
    for (const sec of d.hasMenuSection || []) {
      for (const it of sec.hasMenuItem || []) {
        const offers = Array.isArray(it.offers) ? it.offers : it.offers ? [it.offers] : [];
        if (offers.length > 1) offers.forEach(o => rows.push([`${it.name} – ${o.name}`, Number(o.price), sec.name]));
        else rows.push([it.name, Number(offers[0] ? offers[0].price : it.price), sec.name]);
      }
    }
  }
  return rows.length > 1 ? rows : 'NO MENU FOUND';
}

// Bolt Food: read the public JSON API the food.bolt.eu web app uses (no browser / worker needed).
// Bolt rate-limits bursts (~10 quick calls), so middayRefresh pauses between stores.
const BOLT_API = 'https://deliveryuser.live.boltsvc.net/deliveryClient/public/';
const BOLT_GEO = { lat: 45.6427, lng: 25.5887 };   // Brașov centre: the menu shown depends on the delivery address

function boltGet(endpoint, params) {
  const props = PropertiesService.getScriptProperties();
  let dev = props.getProperty('BOLT_DEVICE');
  if (!dev) { dev = Utilities.getUuid(); props.setProperty('BOLT_DEVICE', dev); }
  const q = Object.assign({ delivery_lat: BOLT_GEO.lat, delivery_lng: BOLT_GEO.lng, version: 'FW.1.120', language: 'ro-RO',
    session_id: dev + 'eater1', distinct_id: '$device:' + dev, country: 'ro', device_name: 'web', device_os_version: 'web',
    deviceId: dev, deviceType: 'web' }, params);
  const url = BOLT_API + endpoint + '?' + Object.entries(q).map(([k, v]) => `${k}=${encodeURIComponent(v)}`).join('&');
  const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true,
    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/125 Safari/537.36',
               'Accept': 'application/json', 'Origin': 'https://food.bolt.eu', 'Referer': 'https://food.bolt.eu/' } });
  if (res.getResponseCode() !== 200) throw new Error('HTTP ' + res.getResponseCode());
  const d = JSON.parse(res.getContentText());
  if (d.code !== 0) throw new Error('Bolt ' + d.code + ' ' + d.message);   // 1005 = TOO_MANY_REQUESTS
  return d.data;
}

function BOLT_MENU(storeUrl) {
  const m = String(storeUrl).match(/\/p\/(\d+)/);
  if (!m) return 'BAD URL – expected .../p/<id>-name/';
  const providerId = m[1];
  const menu = boltGet('getMenuCategories', { provider_id: providerId });
  const items = Object.values(menu.items || {});
  const cats = {};
  items.filter(i => i.type === 'category').forEach(c => cats[c.id] = { name: c.name.value, index: c.index });
  const dishes = items.filter(i => i.type === 'dish' && i.price)
    .sort((a, b) => ((cats[a.parent_id] || {}).index - (cats[b.parent_id] || {}).index) || (a.index - b.index));
  // a dish can be listed under several categories (e.g. "Popular"): keep the first occurrence only
  const rows = [['Item', 'Price', 'Original price', 'Stoc', 'Category']], seen = new Set();
  dishes.forEach(d => {
    const name = d.name.value.replace(/\s+/g, ' ').trim();
    if (seen.has(name)) return;
    seen.add(name);
    const p = d.price, orig = p.original_price && p.original_price.value > p.value ? p.original_price.value : '';
    rows.push([name, p.value, orig, d.availability === 'in_stock' ? '' : 'epuizat', (cats[d.parent_id] || {}).name || '']);
  });
  if (rows.length < 2) return 'NO ITEMS';

  LAST_META = null;
  try {
    Utilities.sleep(1500);
    const p = boltGet('getProviderDetails', { provider_id: providerId, city_id: 802 }).provider;
    const eta = p.min_delivery_eta ? `${Math.round(p.min_delivery_eta / 60)}–${Math.round(p.max_delivery_eta / 60)}` : '';
    LAST_META = { rating: p.rating_info ? p.rating_info.rating : '', reviews: p.rating_info ? p.rating_info.count : '',
      fee: p.delivery_price ? p.delivery_price.value : '', minutes: eta };
  } catch (e) { Logger.log(`${providerId}: meta skipped (${e.message})`); }
  return rows;
}

// Dump the whole competitor menu: =FB_MENU("uid")
function FB_MENU(restaurantUid) {
  const json = UrlFetchApp.fetch('https://www.foodbooking.com/api/restaurant/' + restaurantUid + '/menu',
    { muteHttpExceptions: true }).getContentText();
  const menu = JSON.parse(json);
  if (!menu.categories) return 'NO ITEMS';
  const rows = [['Category', 'ID', 'Item', 'Price']];
  const seen = new Set();
  for (const cat of menu.categories) {
    for (const it of cat.items) {
      rows.push([cat.name, it.id, it.name, it.price]);
      for (const g of it.groups || []) {
        for (const o of g.options || []) {
          if (!(o.price > 0)) continue;                          // free options aren't prices
          const key = `${g.name.trim()} | ${o.name.trim()}`;
          if (seen.has(key)) continue;                           // same extra offered on many items
          seen.add(key);
          rows.push([`Extra: ${g.name.trim()}`, `x:${key}`, key, o.price]);
        }
      }
    }
  }
  return rows;
}


function SITE_MENU(url) {
  const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true,
    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/125 Safari/537.36', 'Accept': 'text/html' } });
  if (res.getResponseCode() !== 200) return 'HTTP ' + res.getResponseCode();
  const html = res.getContentText();
  const strip = t => t.replace(/<[^>]+>/g, '').replace(/&nbsp;|\u00a0/g, ' ').replace(/&amp;/g, '&').replace(/&#8211;|&#8217;/g, "'").replace(/\s+/g, ' ').trim();
  const rows = [['Item', 'Price', 'Source']];

  // 1. catalogue items
  const cat = /av-catalogue-title'>([\s\S]*?)<\/div>\s*<div class='av-catalogue-price'>([\s\S]*?)<\/div>/g;
  let m;
  while ((m = cat.exec(html)) !== null) rows.push([strip(m[1]), parseFloat(strip(m[2]).replace(',', '.')), 'catalog']);

  // 2. featured items: any <h3>NAME</h3> followed by <h4>Label: price</h4> / <h4>price</h4> before the next <h3>
  const parts = html.split(/<h3\b[^>]*>/).slice(1);
  for (const part of parts) {
    const name = strip((part.match(/^([\s\S]*?)<\/h3>/) || [])[1] || '');
    if (!name) continue;
    const h4s = [...part.matchAll(/<h4\b[^>]*>([\s\S]*?)<\/h4>/g)].map(x => strip(x[1]));
    for (const t of h4s) {
      const mm = t.match(/^(?:([^:]+):\s*)?(\d+(?:[.,]\d+)?)\s*(lei|ron)?$/i);
      if (!mm) continue;
      const label = mm[1] ? mm[1].trim() : '';
      rows.push([label && label.toLowerCase() !== 'burger' ? `${name} ${label}` : name, parseFloat(mm[2].replace(',', '.')), 'featured']);
    }
  }
  return rows.length > 1 ? rows : 'NO ITEMS';
}


function WOLT_MENU(venueUrl) {
  venueUrl = String(venueUrl).trim();
  const slug = venueUrl.replace(/\/+$/, '').split('/').pop();
  const url = `https://consumer-api.wolt.com/consumer-api/consumer-assortment/v1/venues/slug/${slug}/assortment?language=ro`;
  const res = UrlFetchApp.fetch(url, { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0', 'Accept': 'application/json' } });
  if (res.getResponseCode() !== 200) return 'HTTP ' + res.getResponseCode();
  const d = JSON.parse(res.getContentText());
  const catOf = {};
  d.categories.forEach(c => (c.item_ids || []).forEach(id => catOf[id] = c.name));
  const rows = [['Item', 'Price', 'Category', 'ID']];
  const seen = {};
  d.items.forEach(it => { const n = it.name.trim(); seen[n] = (seen[n] || 0) + 1; });
  d.items.forEach(it => {
    const n = it.name.trim(), cat = catOf[it.id] || '';
    rows.push([seen[n] > 1 && cat ? `${n} (${cat})` : n, it.price / 100, cat, it.id]);
  });
  return rows;
}
