const OWN_REPORTS = [
  { name: 'Report A – TwoGuys Delivery', map: 'TwoGuys Delivery', tabs: ['TwoGuys - G', 'TwoGuys - W', 'TwoGuys - B', 'Passage by TwoGuys - W'] },
  { name: 'Report B – TwoGuys Home',      map: 'TwoGuys Home', tabs: ['Burgerbrasov', 'Mypassage'] },
  { name: 'Report C – Smash Delivery',   map: 'Smash Delivery', tabs: ['TheSmash - G', 'TheSmash - W', 'TheSmash - B'] },
  // breakfast: two Wolt venues + the breakfast category inside the full-menu venues (those carry many more products, so no "extra" check there)
  { name: 'Report D – Breakfast',        map: 'Breakfast', tabs: ['Br. by Passage - W', 'Br. by TwoGuys - W', 'TwoGuys - W', 'Passage by TwoGuys - W'], noExtras: ['TwoGuys - W', 'Passage by TwoGuys - W'] },
];
const MASTER_HEAD = ['Produs (nume de referință – se editează manual)', 'Preț (de referință – se editează manual)'];
const PRICE_TOLERANCE = 0.01;
const SGR = 0.5;                                   // deposit not shown in Bolt prices
const SGR_TABS = ['TwoGuys - B', 'TheSmash - B'];  // tabs where drinks appear SGR-less

// quantities in a name: "330ml", "250 ml", "450g", "2 buc" -> ["330ml", ...]
function qtyOf(s) {
  return (String(s).toLowerCase().match(/\d+(?:[.,]\d+)?\s*(?:ml|l|g|gr|kg|buc|cl)\b/g) || []).map(q => q.replace(/\s+/g, '').replace('gr', 'g'));
}

// Creates the Ref tabs; column A/B stay yours. Platform columns get fuzzy suggestions for blank cells only.
function buildRefSheets() {
  const ss = SpreadsheetApp.getActive();
  for (const r of OWN_REPORTS) {
    let sh = ss.getSheetByName(r.map);
    if (!sh) { sh = ss.insertSheet(r.map); sh.appendRow([...MASTER_HEAD, ...r.tabs]); sh.setFrozenRows(1); sh.setFrozenColumns(2); }
    if (sh.getLastRow() < 2) continue;
    const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
    const rng = sh.getRange(2, 1, sh.getLastRow() - 1, headers.length), rows = rng.getValues();
    
    headers.forEach((tab, c) => {
      if (!COMPETITORS.some(x => x.sheet === tab)) return;
      const prices = priceOf(ss, tab), pool = Object.keys(prices);
      const used = new Set(rows.map(x => String(x[c])).filter(Boolean));
      for (const row of rows) {
        if (!row[0] || row[c]) continue;
        const base = Number(row[1]), a = mapNorm(row[0]);
        let best = '', bestScore = 0;
        for (const cand of pool) {
          if (used.has(cand)) continue;
          const p = prices[cand];
          if (base && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) continue;
          const qa = qtyOf(row[0]), qb = qtyOf(cand);
          if (qa.length && qb.length && qa.join() !== qb.join()) continue;   // 250ml ≠ 330ml
          const s = similarity(a, mapNorm(cand));
          if (s > bestScore) { bestScore = s; best = cand; }
        }
        if (bestScore >= MAP_THRESHOLD) { row[c] = best; used.add(best); }
      }
    });
    rng.setValues(rows);
  }
}

// Products of one platform tab, in menu order: [{ name, list }] (list price = original price when discounted)
function platformItems(ss, tab) {
  const cfg = COMPETITORS.find(c => c.sheet === tab), sh = ss.getSheetByName(tab), out = [];
  if (!cfg || !sh || sh.getLastRow() < 2) return out;
  const last = sh.getLastRow(), col = c => c ? sh.getRange(`${c}2:${c}${last}`).getValues().flat() : [];
  const n = col(cfg.nameCol), p = col(cfg.priceCol), o = col(cfg.origCol);
  const seen = new Set();
  n.forEach((x, i) => { const s = Number(p[i]), org = Number(o[i]), name = String(x || '').trim(); if (name && s > 0 && !seen.has(name)) { seen.add(name); out.push({ name, list: org > s ? org : s }); } });
  return out;
}

// One-time: rebuild the delivery master tabs (TwoGuys Delivery, Smash Delivery) from the platforms.
// Column A/B = Glovo name and list price; products that exist only on Wolt/Bolt are appended with their own name/price,
// so they show up as missing on the other platforms. Platform columns are auto-matched like buildRefSheets.
// REPLACES the current content of those tabs.
function rebuildDeliveryMasters() {
  const ss = SpreadsheetApp.getActive();
  const targets = OWN_REPORTS.filter(r => r.tabs.some(t => / - G$/.test(t)));
  for (const r of targets) {
    const sh = ss.getSheetByName(r.map) || ss.insertSheet(r.map);
    const gTab = r.tabs.find(t => / - G$/.test(t));
    const rows = platformItems(ss, gTab).map(it => [it.name, it.list, ...r.tabs.map(t => t === gTab ? it.name : '')]);
    sh.clearContents();
    sh.getRange(1, 1, 1, 2 + r.tabs.length).setValues([[...MASTER_HEAD, ...r.tabs]]).setFontWeight('bold');
    if (rows.length) sh.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
    sh.setFrozenRows(1); sh.setFrozenColumns(2);
  }
  buildRefSheets();                                           // fuzzy-match Wolt/Bolt names onto the Glovo rows
  for (const r of targets) {                                  // products only on Wolt/Bolt → own rows
    const sh = ss.getSheetByName(r.map);
    const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
    const data = sh.getLastRow() > 1 ? sh.getRange(2, 1, sh.getLastRow() - 1, headers.length).getValues() : [];
    const extra = [];
    r.tabs.forEach(tab => {
      const c = headers.indexOf(tab); if (c < 0) return;
      const used = new Set(data.map(row => String(row[c]).trim()).concat(extra.map(row => String(row[c]).trim())));
      platformItems(ss, tab).filter(it => !used.has(it.name)).forEach(it => {
        const row = [it.name, it.list, ...r.tabs.map(() => '')]; row[c] = it.name; extra.push(row);
      });
    });
    if (extra.length) sh.getRange(sh.getLastRow() + 1, 1, extra.length, headers.length).setValues(extra);
    Logger.log(`${r.map}: ${data.length} products from ${r.tabs.find(t => / - G$/.test(t))}, +${extra.length} only on other platforms`);
  }
  buildRefSheets();                                           // match the appended rows on the remaining platform
}

// One-time: rebuild "TwoGuys Home" from the restaurant site. Column A/B = Burgerbrasov name and price; the Mypassage column is
// filled by fuzzy match, then by the AI matcher (marked ≈ until you review), and set to "-" where nothing matches
// (not every restaurant product is sold for pick-up). Mypassage-only products are appended as extra rows.
function rebuildHomeMaster() {
  const ss = SpreadsheetApp.getActive();
  const r = OWN_REPORTS.find(x => x.map === 'TwoGuys Home');
  const base = r.tabs[0], other = r.tabs.slice(1);
  const sh = ss.getSheetByName(r.map) || ss.insertSheet(r.map);
  const rows = platformItems(ss, base).map(it => [it.name, it.list, it.name, ...other.map(() => '')]);
  sh.clearContents();
  sh.getRange(1, 1, 1, 2 + r.tabs.length).setValues([[...MASTER_HEAD, ...r.tabs]]).setFontWeight('bold');
  if (rows.length) sh.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
  sh.setFrozenRows(1); sh.setFrozenColumns(2);
  buildRefSheets();                                 // fuzzy pass
  llmFillRef(r.map);                                // AI pass for what is left, marked ≈
  const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
  const rng = sh.getRange(2, 1, sh.getLastRow() - 1, headers.length), data = rng.getValues();
  const extra = [];
  other.forEach(tab => {
    const c = headers.indexOf(tab);
    const used = new Set(data.map(row => String(row[c]).replace(/^[~≈]/, '').trim()));
    platformItems(ss, tab).filter(it => !used.has(it.name)).forEach(it => { const row = [it.name, it.list, ...r.tabs.map(() => '')]; row[c] = it.name; extra.push(row); });
    data.forEach(row => { if (!String(row[c]).trim()) row[c] = '-'; });   // restaurant-only product: not expected on Mypassage
  });
  rng.setValues(data);
  if (extra.length) sh.getRange(sh.getLastRow() + 1, 1, extra.length, headers.length).setValues(extra);
  Logger.log(`${r.map}: ${data.length} products from ${base}, +${extra.length} only on ${other.join('/')}`);
}

// AI matcher for a master tab: fills blank platform cells with the equivalent product name, prefixed ≈ (review, then remove the ≈).
function llmFillRef(mapName) {
  const ss = SpreadsheetApp.getActive();
  const key = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_KEY');
  if (!key) { Logger.log('ANTHROPIC_KEY not set – AI pass skipped'); return; }
  const sh = ss.getSheetByName(mapName);
  const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
  const rng = sh.getRange(2, 1, sh.getLastRow() - 1, headers.length), rows = rng.getValues().map(r => r.map(String));
  headers.forEach((tab, c) => {
    if (c < 2 || !COMPETITORS.some(x => x.sheet === tab)) return;
    const blanks = rows.filter(r => r[0] && !r[c].trim());
    if (!blanks.length) return;
    const taken = new Set(rows.map(r => r[c].replace(/^[~≈]/, '').trim()).filter(Boolean));
    const candidates = platformItems(ss, tab).filter(it => !taken.has(it.name));
    if (!candidates.length) return;
    const prompt = `Two sales channels of the SAME restaurant list the same dishes under different spellings or formats
(capitalisation, diacritics, "Meniu X" vs "X – Meniu", weight suffixes, "Extra: bacon" vs "Extra bacon", Romanian vs English).
Match each product from list A to the SAME product in list B. Only match when it is clearly the same dish and the same portion/variant
(a burger alone ≠ its menu/combo; "single" ≠ "double"). Return null when there is no equivalent. Each list-B item may be used at most once.

LIST A (name | price RON):
${blanks.map(r => `${r[0]} | ${r[1] || '?'}`).join('\n')}

LIST B "${tab}" (name | price RON):
${candidates.map(it => `${it.name} | ${it.list}`).join('\n')}

Respond with ONLY a JSON object, no prose, no markdown: {"<list A name>": "<list B name exactly as written>" or null, ...}. Include every list A item as a key.`;
    let res;
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
          method: 'post', contentType: 'application/json', muteHttpExceptions: true,
          headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' },
          payload: JSON.stringify({ model: LLM_MODEL, max_tokens: 8000, messages: [{ role: 'user', content: prompt }] }),
        });
        if (res.getResponseCode() < 500) break;
      } catch (e) { if (attempt === 3) { Logger.log(`${tab}: network error ${e.message}`); return; } }
      Utilities.sleep(2000 * attempt);
    }
    const body = JSON.parse(res.getContentText());
    if (res.getResponseCode() !== 200 || !body.content) { Logger.log(`${tab}: API ${res.getResponseCode()} ${res.getContentText().slice(0, 300)}`); return; }
    const text = body.content.map(b => b.text || '').join('');
    let out; try { out = JSON.parse(text.replace(/```json|```/g, '').trim()); } catch (e) { Logger.log(`${tab}: bad JSON: ${text.slice(0, 200)}`); return; }
    const valid = new Set(candidates.map(it => it.name)), used = new Set();
    let filled = 0;
    rows.forEach(r => { const v = out[r[0]]; if (r[c].trim() || !v || !valid.has(v) || used.has(v)) return; r[c] = LLM_MARKER + v; used.add(v); filled++; });
    Logger.log(`${mapName} / ${tab}: AI matched ${filled} of ${blanks.length}`);
  });
  rng.setValues(rows);
}

// Adds any platform column that a master tab is missing (e.g. a newly tracked venue) and fills it by fuzzy match. Safe to re-run.
function syncMasterColumns() {
  const ss = SpreadsheetApp.getActive();
  for (const r of OWN_REPORTS) {
    const sh = ss.getSheetByName(r.map); if (!sh || sh.getLastRow() < 1) continue;
    const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
    r.tabs.filter(t => !headers.includes(t)).forEach(t => { headers.push(t); sh.getRange(1, headers.length).setValue(t).setFontWeight('bold'); Logger.log(r.map + ': added column ' + t); });
    sh.getRange(1, 1, 1, 2).setValues([MASTER_HEAD]).setFontWeight('bold');
  }
  buildRefSheets();
}

// Self-maintaining master: when a platform renames a product, the master cell that pointed at the old name follows it;
// blank cells are filled the same way buildRefSheets does. Returns true if any cell changed. Nothing is ever removed.
function syncMasterNames(r, rows, tabs, tabIdx, prices, renamed) {
  let changed = false;
  tabs.forEach((tab, i) => {
    const c = tabIdx[i], pool = Object.keys(prices[i]);
    const used = new Set(rows.map(row => String(row[c] || '').replace(/^[~≈]/, '').trim()).filter(Boolean));
    for (const row of rows) {
      const name = String(row[0]).trim(); if (!name) continue;
      const raw = String(row[c] || '').trim(), mapped = raw.replace(/^[~≈]/, '').trim();
      if (mapped === '-' || (mapped && prices[i][mapped])) continue;              // n/a or still valid
      if (mapped && CFG().scheduledItem(mapped)) continue;                         // off today by schedule, not renamed
      const base = Number(row[1]), ref = mapped || name;                           // match against the old platform name, else our name
      let best = '', bestScore = 0;
      for (const cand of pool) {
        if (used.has(cand)) continue;
        if (cosmeticKey(cand) === cosmeticKey(ref)) { best = cand; bestScore = 1; break; }
        const p = prices[i][cand].list;
        if (base && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) continue;
        const qa = qtyOf(ref), qb = qtyOf(cand);
        if (qa.length && qb.length && qa.join() !== qb.join()) continue;
        const s = similarity(mapNorm(ref), mapNorm(cand));
        if (s > bestScore) { bestScore = s; best = cand; }
      }
      // replacing a known name needs a close match — unless the candidate has exactly the master price (± SGR on Bolt)
      const samePrice = best && base && Math.abs(prices[i][best].list - base) <= PRICE_TOLERANCE + (SGR_TABS.includes(tab) ? SGR : 0);
      const need = mapped && !samePrice ? RENAME_THRESHOLD : MAP_THRESHOLD;
      if (best && bestScore >= need) {
        if (mapped) renamed.push({ map: r.map, tab, from: mapped, to: best });
        row[c] = best; used.add(best); changed = true;
      }
    }
  });
  return changed;
}

function ownShopsReport() {
  const ss = SpreadsheetApp.getActive(), today = Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd');
  const listPrices = tab => {                       // name -> { list price (discount ignored), disc, oos, cat, pos }
    const cfg = COMPETITORS.find(c => c.sheet === tab), sh = ss.getSheetByName(tab), m = {};
    if (!cfg || !sh || sh.getLastRow() < 2) return m;
    const last = sh.getLastRow(), col = c => c ? sh.getRange(`${c}2:${c}${last}`).getValues().flat() : [];
    const n = col(cfg.nameCol), p = col(cfg.priceCol), o = col(cfg.origCol), st = col(cfg.stockCol), ct = col(cfg.catCol), pr = col(cfg.promoCol);
    n.forEach((x, i) => { const s = Number(p[i]), org = Number(o[i]), badge = promoPct(pr[i]);
      if (x && s > 0) m[String(x).trim()] = { list: org > s ? org : s, disc: org > s ? (badge > 0 ? badge : Math.round((1 - s / org) * 100)) : 0, oos: /epuizat/i.test(st[i] || ''), cat: String(ct[i] || '').trim(), pos: i }; });
    return m;
  };
  const esc = s => String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  const col = EMAIL_CFG.colors;
  const platform = tab => displayName(tab).replace(/.* - /, '');
  let html = '', total = 0;
  const tg = [];                                    // plain-text lines for Telegram
  // categories remembered by the daily snapshot, so a product that is off today (breakfast, weekend-only) still has one
  const snapCat = {};
  const snap = ss.getSheetByName(SNAP_SHEET);
  if (snap && snap.getLastRow() > 1) snap.getRange(2, 1, snap.getLastRow() - 1, 8).getValues().forEach(r => { if (r[7]) (snapCat[r[0]] = snapCat[r[0]] || {})[String(r[2]).trim()] = String(r[7]); });
  const renamed = [];                               // master cells updated automatically this run

  for (const r of OWN_REPORTS) {
    const map = ss.getSheetByName(r.map); if (!map || map.getLastRow() < 2) continue;
    const headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String).filter(Boolean);
    const tabs = headers.filter(h => COMPETITORS.some(c => c.sheet === h));
    const tabIdx = tabs.map(t => headers.indexOf(t));
    const rowsRange = map.getRange(2, 1, map.getLastRow() - 1, headers.length), rows = rowsRange.getValues();
    const prices = tabs.map(listPrices);
    if (syncMasterNames(r, rows, tabs, tabIdx, prices, renamed)) rowsRange.setValues(rows);   // platform renamed a product → follow it
    const mappedPerTab = tabs.map(() => new Set());
    const issues = [], extras = [];                 // { cat, pos, name, html, tg }
    let nameDiffs = 0;
    // category of a master row = category on the first platform where it is mapped; fallback: "CAT | Name" prefix in the master name
    const catOf = (row, name) => {
      for (let i = 0; i < tabs.length; i++) { const p = prices[i][String(row[tabIdx[i]] || '').trim()]; if (p && p.cat) return { cat: p.cat, pos: p.pos }; }
      for (let i = 0; i < tabs.length; i++) { const c = (snapCat[tabs[i]] || {})[String(row[tabIdx[i]] || '').replace(/^[~≈]/, '').trim()]; if (c) return { cat: c, pos: 9000 }; }
      const m = name.match(/^([^|]+)\|/); return { cat: m ? m[1].trim() : 'Altele', pos: 9999 };
    };

    for (const row of rows) {
      const name = String(row[0]).trim(), refP = Number(row[1]);
      if (!name || !(refP > 0)) continue;
      const cells = [], notes = [], rowCat = catOf(row, name); let bad = false;
      tabs.forEach((tab, i) => {
        const raw = String(row[tabIdx[i]] || '').trim(), unreviewed = /^[~≈]/.test(raw), mapped = raw.replace(/^[~≈]/, '').trim();
        if (mapped === '-') { cells.push('<td align="center" style="color:#9ca3af;">n/a</td>'); return; }
        if (mapped) mappedPerTab[i].add(mapped);
        const p = mapped ? prices[i][mapped] : undefined;
        if (p == null && (CFG().scheduledItem(name) || CFG().scheduledItem(mapped) || CFG().scheduledTabs.includes(tab) || CFG().scheduledCategory(rowCat.cat))) { cells.push('<td align="center" style="color:#9ca3af;" title="produs cu program">program</td>'); return; }
        if (p == null) { cells.push(`<td align="center" style="color:${col.up};font-weight:700;">X</td>`); notes.push(`${platform(tab)}: lipsă`); bad = true; return; }
        const tag = (unreviewed ? ' <span style="color:#d97706;font-size:11px;" title="potrivire AI nerevizuită">≈</span>' : '') + (p.disc ? ` <span style="color:${col.ADDED};font-size:11px;">−${p.disc}%</span>` : '')
                  + (p.oos ? ` <span style="color:${col.REMOVED};font-size:11px;">epuizat</span>` : '');
        const nameDiff = cosmeticKey(mapped) !== cosmeticKey(name);
        const nameTag = nameDiff ? ` <span style="color:#d97706;font-size:11px;" title="numele de pe platformă diferă de lista master">nume: ${esc(mapped)}</span>` : '';
        if (nameDiff) { nameDiffs++; notes.push(`${platform(tab)}: nume „${mapped}”`); bad = true; }
        let d = p.list - refP;
        const sgr = SGR_TABS.includes(tab) && Math.abs(d + SGR) <= PRICE_TOLERANCE;
        if (sgr) d = 0;
        if (Math.abs(d) > PRICE_TOLERANCE) { cells.push(`<td align="right" style="color:${d > 0 ? col.up : col.down};font-weight:700;">${p.list.toFixed(2)} (${d > 0 ? '+' : ''}${d.toFixed(2)})${tag}${nameTag}</td>`); notes.push(`${platform(tab)}: ${p.list.toFixed(2)} (${d > 0 ? '+' : ''}${d.toFixed(2)})`); bad = true; }
        else cells.push(`<td align="center" style="color:#9ca3af;">✓${sgr ? ' <span style="font-size:11px;">SGR</span>' : ''}${tag}${nameTag}</td>`);
      });
      if (bad) issues.push({ ...rowCat, name, html: `<tr><td>${esc(name)}</td><td align="right">${refP.toFixed(2)}</td>${cells.join('')}</tr>`,
        tg: `• ${tgEsc(name)} [${refP.toFixed(2)}] — ${tgEsc(notes.join(', '))}` });
    }

    // items on a platform that no master row points at → add them to the master tab, or delete them from the platform
    tabs.forEach((tab, i) => (r.noExtras || []).includes(tab) ? null : Object.keys(prices[i]).filter(n => !mappedPerTab[i].has(n)).forEach(n => {
      const p = prices[i][n];
      extras.push({ cat: p.cat || 'Altele', pos: p.pos, name: n, tab,
        html: `<tr><td>${esc(platform(tab))}</td><td>${esc(n)}</td><td align="right">${p.list.toFixed(2)}</td></tr>`,
        tg: `• ${tgEsc(n)} (${p.list.toFixed(2)}) e pe ${platform(tab)} dar nu în „${tgEsc(r.map)}”` });
    }));

    // group by category, in platform menu order; rows inside a category alphabetically
    const catOrder = {};
    [...issues, ...extras].forEach(x => catOrder[x.cat] = Math.min(catOrder[x.cat] == null ? Infinity : catOrder[x.cat], x.pos));
    const byCat = (a, b) => (catOrder[a.cat] - catOrder[b.cat]) || a.cat.localeCompare(b.cat) || a.name.localeCompare(b.name);
    issues.sort(byCat); extras.sort((a, b) => byCat(a, b) || a.tab.localeCompare(b.tab));
    const withHeaders = (list, span, fn) => { let last = null, out = ''; list.forEach(x => {
      if (x.cat !== last) { out += `<tr><td colspan="${span}" style="background:#f3f4f6;font-weight:700;color:#374151;">${esc(x.cat)}</td></tr>`; last = x.cat; }
      out += fn(x); }); return out; };
    const tgWithHeaders = list => { let last = null; const out = []; list.forEach(x => { if (x.cat !== last) { out.push(`<i>${tgEsc(x.cat)}</i>`); last = x.cat; } out.push(x.tg); }); return out; };

    total += issues.length + extras.length;
    if (issues.length || extras.length) tg.push(`\n<b>${tgEsc(r.name)}</b>`, ...tgWithHeaders(issues));
    if (extras.length) tg.push(`<i>Pe platformă dar nu în „${tgEsc(r.map)}”:</i>`, ...extras.map(x => x.tg));
    html += `<div style="font-size:15px;font-weight:700;margin:18px 0 6px;">${esc(r.name)} <span style="font-weight:400;color:#6b7280;font-size:13px;">— ${issues.length} probleme${nameDiffs ? ` (din care ${nameDiffs} nume diferite)` : ''}, ${extras.length} produse în plus</span></div>`;
    html += issues.length ? `<table width="100%" cellpadding="6" cellspacing="0" style="border:1px solid #e5e7eb;font-size:13px;">
      <tr style="background:${col.headerBg};color:#fff;"><th align="left">Produs</th><th align="right">Preț ref.</th>${tabs.map(h => `<th>${esc(platform(h))}</th>`).join('')}</tr>${withHeaders(issues, tabs.length + 2, x => x.html)}</table>` : '';
    html += extras.length ? `<div style="font-size:13px;font-weight:700;margin:10px 0 4px;color:#6b7280;">Pe platformă dar nu în lista master — de verificat tabul „${esc(r.map)}” din fișier:</div>
      <table width="100%" cellpadding="6" cellspacing="0" style="border:1px solid #e5e7eb;font-size:13px;"><tr style="background:${col.headerBg};color:#fff;"><th align="left">Platformă</th><th align="left">Produs</th><th align="right">Preț</th></tr>${withHeaders(extras, 3, x => x.html)}</table>` : '';
    if (!issues.length && !extras.length) html += `<p style="font-size:13px;color:${col.ADDED};">✓ Totul în ordine</p>`;
  }
  if (renamed.length) html += `<div style="font-size:13px;font-weight:700;margin:18px 0 4px;color:#6b7280;">Nume actualizate automat în lista master (platforma a redenumit produsul):</div>
      <table width="100%" cellpadding="6" cellspacing="0" style="border:1px solid #e5e7eb;font-size:13px;"><tr style="background:${col.headerBg};color:#fff;"><th align="left">Tab</th><th align="left">Platformă</th><th align="left">Vechi</th><th align="left">Nou</th></tr>
      ${renamed.map(x => `<tr><td>${esc(x.map)}</td><td>${esc(platform(x.tab))}</td><td>${esc(x.from)}</td><td>${esc(x.to)}</td></tr>`).join('')}</table>`;
  if (!total && !renamed.length) return;
  if (CFG().telegramOwn) tgSend(`🏪 <b>Magazine proprii: ${total} probleme</b> (${today})\n` + tg.join('\n'));
  const body = `<div style="background:#f3f4f6;padding:20px 8px;font-family:Arial,sans-serif;"><div style="max-width:720px;margin:0 auto;background:#fff;border-radius:12px;border:1px solid #e5e7eb;overflow:hidden;">
    <div style="height:5px;background:${col.accent};"></div><div style="padding:20px;text-align:center;border-bottom:1px solid #e5e7eb;"><div style="font-size:18px;font-weight:800;">Verificare magazine proprii</div><div style="font-size:13px;color:#6b7280;">${today} · X = lipsă · preț afișat = diferit de lista master · „nume:” = produsul e pe platformă cu alt nume decât în lista master</div></div>
    <div style="padding:4px 20px 20px;">${html}</div><div style="height:5px;background:${col.accent};"></div></div></div>`;
  MailApp.sendEmail({ to: CFG().ownEmail, subject: `🏪 Magazine proprii: ${total} probleme (${today})`, htmlBody: body });
}
