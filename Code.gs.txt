function FB_PRICE(restaurantUid, itemId) {
  const key = 'fb_' + restaurantUid;
  const cache = CacheService.getScriptCache();
  let json = cache.get(key);
  if (!json) {
    json = UrlFetchApp.fetch('https://www.foodbooking.com/api/restaurant/' + restaurantUid + '/menu').getContentText();
    cache.put(key, json, 1800); // 30 min – don't hammer their API
  }
  const menu = JSON.parse(json);
  for (const cat of menu.categories) {
    for (const it of cat.items) {
      if (String(it.id) === String(itemId)) return it.price;
    }
  }
  return 'NOT FOUND';
}

function registryRows_() {
  const sh = SpreadsheetApp.getActive().getSheetByName(REGISTRY_SHEET);
  if (!sh || sh.getLastRow() < 2) return [];
  const out = [];
  sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues().forEach((v, i) => {
    const brand = String(v[0] || '').trim(); if (!brand) return;
    const on = v[1] === true || /^(da|yes|1|true)$/i.test(String(v[1]).trim());
    Object.keys(REGISTRY_PLATFORMS).forEach(p => { const url = String(v[REGISTRY_PLATFORMS[p].col - 1] || '').trim(); if (url) out.push({ row: i + 2, brand, p, url, on, tab: `${brand} - ${p}` }); });
    const site = String(v[5] || '').trim();
    // older competitor site tabs are named just by the brand ("Marcos", "Vivo"), newer ones "<Brand> - Site"
    if (site) out.push({ row: i + 2, brand, p: 'Site', url: site, on, tab: COMPETITORS.some(c => c.sheet === brand) ? brand : `${brand} - Site` });
  });
  return out;
}

// orange rows = our own shops, light green = competitors (re-applied after every sync, so new rows are coloured too)
function styleCompetitorRows_(reg) {
  const maxR = reg.getMaxRows();
  if (maxR < 2) return;
  const own = new Set(ownBrands_());
  const brands = reg.getRange(2, 1, maxR - 1, 1).getValues().flat().map(b => String(b).trim());
  reg.getRange(2, 1, maxR - 1, 6).setBackgrounds(brands.map(b => Array(6).fill(own.has(b) ? OWN_BRAND_COLOR : HL_COLORS_REG.tintEdit)));
  reg.getRange(2, 1, maxR - 1, 1).setFontWeights(brands.map(b => [own.has(b) ? 'bold' : 'normal']));
}

// One-time: fills the Competitors tab with every shop already tracked in the COMPETITORS list (our own shops first, in orange), one row per
// brand, with its Glovo / Wolt / Bolt links and website. Brands already in the tab are left alone. Safe to re-run.
function populateCompetitors() {
  const ss = SpreadsheetApp.getActive();
  setupCompetitors();
  const reg = ss.getSheetByName(REGISTRY_SHEET);
  // the tab has tick boxes down to row 501, so getLastRow() is useless here: collect the filled rows by their Brand cell
  const maxR = reg.getMaxRows(), existing = maxR > 1 ? reg.getRange(2, 1, maxR - 1, REGISTRY_COLS).getValues().filter(r => String(r[0]).trim()) : [];
  const have = new Set(existing.map(r => String(r[0]).trim()));
  const brands = {};                                                       // brand -> [Activ, Glovo, Wolt, Bolt, Website]
  COMPETITORS.forEach(c => {
    const m = c.sheet.match(/^(.*) - (G|W|B)$/), brand = m ? m[1] : c.sheet;
    const row = brands[brand] = brands[brand] || [true, '', '', '', ''];
    let url = c.url || '';
    if (!c.fn) { const sh = ss.getSheetByName(c.sheet); const a1 = sh ? String(sh.getRange(1, 1).getValue()) : ''; url = /^https?:/.test(a1) ? a1 : ''; }   // IMPORTXML tabs keep their page in A1
    row[m ? { G: 1, W: 2, B: 3 }[m[2]] : 4] = url || '(adresa e în tab)';
  });
  const add = Object.entries(brands).filter(([b]) => !have.has(b)).sort((a, b) => a[0].localeCompare(b[0])).map(([b, r]) => [b, ...r, '']);
  const ownOrder = ownBrands_().filter((b, i, a) => a.indexOf(b) === i), rank = r => { const i = ownOrder.indexOf(String(r[0]).trim()); return i === -1 ? 999 : i; };
  const all = existing.concat(add).sort((x, y) => rank(x) - rank(y));       // our shops on top (stable sort keeps the rest in order)                                        // existing rows first (this also pulls back rows an earlier run wrote far down), then the new brands
  if (all.length) {
    if (maxR < all.length + 1) reg.insertRowsAfter(maxR, all.length + 1 - maxR);
    const body = reg.getRange(2, 1, reg.getMaxRows() - 1, REGISTRY_COLS);
    body.clearContent(); reg.getRange(2, 2, reg.getMaxRows() - 1, 1).clearDataValidations();
    reg.getRange(2, 1, all.length, REGISTRY_COLS).setValues(all);
    reg.getRange(2, 2, 500, 1).insertCheckboxes().setHorizontalAlignment('center');                       // keeps the ticked ones ticked
    if (reg.getMaxRows() > 501) reg.deleteRows(502, reg.getMaxRows() - 501);
    styleCompetitorRows_(reg);
  }
  Logger.log(`Competitors: ${add.length} branduri adăugate, ${Object.keys(brands).length - add.length} existau deja`);
  syncCompetitors();                                                       // fills the Status column
}

// Creates the tab for every new link and fetches its menu once, so tomorrow's run already compares against a baseline.
function syncCompetitors() {
  const ss = SpreadsheetApp.getActive(), reg = ss.getSheetByName(REGISTRY_SHEET);
  if (!reg) return;
  const status = {};                                               // row -> [lines]
  const say = (row, t) => (status[row] = status[row] || []).push(t);
  let added = false;
  registryRows_().forEach(r => {
    if (!r.on && !ownTab_(r.tab)) return say(r.row, `${r.p}: oprit`);
    let sh = ss.getSheetByName(r.tab), cfg = COMPETITORS.find(c => c.sheet === r.tab);
    if (sh && sh.getLastRow() > 1 && cfg) return say(r.row, `${r.p}: urmărit (${sh.getLastRow() - 1} produse)`);
    const created = !sh;
    if (created) sh = ss.insertSheet(r.tab);
    let data; LAST_META = null;
    if (r.p === 'Site') {                                          // try each known site reader until one finds a menu
      for (let i = 0; i < REGISTRY_SITE_READERS.length && !(Array.isArray(data) && data.length > 1); i++) {
        try { data = this[REGISTRY_SITE_READERS[i].fn](r.url); } catch (e) { data = 'ERR ' + e.message; }
        if (Array.isArray(data) && data.length > 1) {
          PropertiesService.getScriptProperties().setProperty('site_reader_' + r.tab, String(i));
          cfg = { sheet: r.tab, url: r.url, ...REGISTRY_SITE_READERS[i] };
          if (!COMPETITORS.some(c => c.sheet === r.tab)) COMPETITORS.push(cfg);
        }
      }
      if (!(Array.isArray(data) && data.length > 1)) {
        if (created) ss.deleteSheet(sh);
        return say(r.row, 'Site: nu știu să citesc acest site (nu e WooCommerce / meniu schema.org) – are nevoie de un cititor dedicat');
      }
    } else {
      cfg = cfg || { sheet: r.tab, url: r.url, ...registryCfg_(r.p) };
      try { data = this[cfg.fn](cfg.url); } catch (e) { data = 'ERR ' + e.message; }
    }
    if (Array.isArray(data) && data.length > 1) {
      writeDataTab_(sh, data); writeMeta(ss, r.tab); styleDataTab_(sh);
      say(r.row, `${r.p}: adăugat ${data.length - 1} produse`); added = true;
    } else say(r.row, `${r.p}: nu s-au putut citi produsele (${String(data).slice(0, 50)}) – se reîncearcă`);
    Utilities.sleep(r.p === 'B' ? 5000 : 1000);
  });
  // per brand: which platforms have no link at all - a quick way to spot a shop you forgot
  const cells = reg.getRange(2, 1, reg.getMaxRows() - 1, 6).getValues();
  Object.keys(status).forEach(row => {
    const v = cells[Number(row) - 2] || [], miss = ['Glovo', 'Wolt', 'Bolt', 'Website'].filter((n, i) => !String(v[2 + i] || '').trim());
    if (miss.length) status[row].push('fără: ' + miss.join(', '));
  });
  Object.entries(status).forEach(([row, lines]) => reg.getRange(Number(row), 7).setValue(lines.join(' · ')));
  styleCompetitorRows_(reg);
  // new competitor tab(s): add their column to CompMap (fuzzy ~ suggestions to review; existing cells untouched) and refresh MatchMatrix
  if (added) {
    try { prefillCompMapNew(); annotateCompMap(); buildMatrix(); orderTabs(); } catch (e) { Logger.log('CompMap/MatchMatrix not updated: ' + e.message); }
  }
}

let LAST_META = null;

function refreshSources() {
  const ss = SpreadsheetApp.getActive();
  const status = {};                       // sheet -> error text ('' = ok)
  try { syncCompetitors(); } catch (e) { Logger.log('syncCompetitors: ' + e.message); }   // new rows in the Competitors tab
  for (const cfg of COMPETITORS) {
    const sh = ss.getSheetByName(cfg.sheet);
    if (!sh) continue;
    if (cfg.midday || cfg.evening) { status[cfg.sheet] = ''; continue; }   // refreshed by middayRefresh() / eveningRefresh()
    if (cfg.fn && cfg.url) {
      let data, best;
      const have = Math.max(sh.getLastRow() - 1, 0), partial = d => !CFG().scheduledTabs.includes(cfg.sheet) && have >= 10 && d.length - 1 < have * 0.5;
      for (let attempt = 0; attempt < 3; attempt++) {                // Glovo / Wolt / the site sometimes answer 403/429/timeout once: retry after a pause
        if (attempt) Utilities.sleep(attempt * 4000);
        LAST_META = null;
        try { data = this[cfg.fn](cfg.url); } catch (e) { data = 'ERR ' + e.message; }
        if (Array.isArray(data) && data.length > 1) {
          if (!partial(data)) break;                                 // a half-loaded menu (e.g. 43 of 104) counts as a failed read: retry
          if (!best || data.length > best.length) best = data;
          data = 'PARTIAL';
        }
      }
      if (data === 'PARTIAL') data = `${best.length - 1} produse față de ${have} ieri – tab păstrat neschimbat`;   // yesterday's data stays in the tab, dailyCheck flags it
      if (Array.isArray(data) && data.length > 1) {
        writeDataTab_(sh, data);
        writeMeta(ss, cfg.sheet);                                   // Glovo / Wolt rating & fee (Bolt is written by middayRefresh)
        status[cfg.sheet] = '';
      } else {
        status[cfg.sheet] = String(data);   // e.g. HTTP 404, NO ITEMS
      }
    } else {
      const rng = sh.getRange(1, 1, 3, 10);
      rng.getFormulas().forEach((row, r) => row.forEach((formula, c) => {
        if (/IMPORT(XML|DATA|HTML)/i.test(formula)) {
          rng.getCell(r + 1, c + 1).setFormula('');
          SpreadsheetApp.flush();
          rng.getCell(r + 1, c + 1).setFormula(formula);
        }
      }));
      status[cfg.sheet] = '';
    }
  }
  
  SpreadsheetApp.flush();
  const importTabs = COMPETITORS.filter(c => !c.fn);
  for (let i = 0; i < 18; i++) {              // up to 18 × 10s = 3 min
    Utilities.sleep(10000);
    SpreadsheetApp.flush();
    const pending = importTabs.filter(c => {
      const sh = ss.getSheetByName(c.sheet); if (!sh) return false;
      const vals = sh.getRange(`${c.priceCol}2:${c.priceCol}${Math.max(sh.getLastRow(), 2)}`).getValues().flat();
      return !vals.some(v => typeof v === 'number' && v > 0);
    });
    if (!pending.length) break;
  }
  return status;
}
// Writes a reader's rows into its tab: the old content goes first, so a tab still in an older layout is replaced completely.
// Column D (Discount %) is a plain number: a percent format left over from an older layout would show 15 as 1500%.
function writeDataTab_(sh, data) {
  sh.getRange(1, 1, sh.getMaxRows(), data[0].length).clearContent();
  sh.getRange(1, 1, data.length, data[0].length).setValues(data);
  if (data[0].length >= 4) sh.getRange(2, 4, Math.max(sh.getMaxRows() - 1, 1), 1).setNumberFormat('0');
}

// One-time, right after pasting the standard-layout version: rewrites every platform / website tab in the new layout (Bolt tabs too,
// which dailyCheck leaves to middayRefresh). Tabs already converted are skipped, so if the 6-minute limit stops it, run it again.
function migrateLayout() {
  const ss = SpreadsheetApp.getActive();
  let done = 0, todo = 0;
  for (const cfg of COMPETITORS) {
    if (!STD_FN.test(cfg.fn || '') || !cfg.url) continue;
    const sh = ss.getSheetByName(cfg.sheet); if (!sh) continue;
    if (!oldLayoutTab_(sh, cfg)) { Logger.log(`✓ ${cfg.sheet}: already in the new layout`); continue; }
    LAST_META = null;
    let data; try { data = this[cfg.fn](cfg.url); } catch (e) { data = 'ERR ' + e.message; }
    if (Array.isArray(data) && data.length > 1) { writeDataTab_(sh, data); done++; Logger.log(`✅ ${cfg.sheet}: ${data.length - 1} produse`); }
    else { todo++; Logger.log(`❌ ${cfg.sheet}: ${String(data).slice(0, 60)} – run migrateLayout again`); }
    Utilities.sleep(cfg.midday ? 5000 : 1000);                       // Bolt rate-limits bursts
  }
  Logger.log(`migrateLayout: ${done} tabs converted, ${todo} to retry`);
}

function CFG() {
  if (_cfg) return _cfg;
  const raw = {};
  SETTINGS_DEFAULTS.forEach(([k, v]) => raw[k] = v);
  const sh = SpreadsheetApp.getActive().getSheetByName(SETTINGS_SHEET);
  if (sh && sh.getLastRow() > 1) {
    sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(([k, v]) => { if (k in raw && v !== '' && v != null) raw[k] = v; });
  }
  const list = s => String(s || '').split(',').map(x => x.trim()).filter(Boolean);
  const num = (v, d) => isNaN(Number(v)) ? d : Number(v);
  const rules = s => list(s).map(x => x.endsWith('*') ? { has: x.slice(0, -1).toLowerCase() } : { is: x.toLowerCase() });
  const matches = (rs, name) => { const n = String(name).toLowerCase(); return rs.some(r => r.is ? n === r.is : n.includes(r.has)); };
  const ignore = rules(raw.ignore_items), scheduled = rules(raw.scheduled_items), schedCats = rules(raw.scheduled_categories);
  _cfg = {
    alertEmail: list(raw.alert_email).join(',') || ALERT_EMAIL,
    ownEmail: list(raw.own_email).join(',') || ALERT_EMAIL,
    weeklyEmail: list(raw.weekly_email).join(',') || ALERT_EMAIL,
    metaEmail: list(raw.meta_email).join(',') || ALERT_EMAIL,
    missingRuns: Math.max(1, num(raw.missing_runs, 3)),
    minDiffLei: num(raw.min_diff_lei, 0),
    minDiffPct: num(raw.min_diff_pct, 0) / 100,
    forgetRemovedDays: num(raw.forget_removed_days, 60),
    collapseAfter: num(raw.collapse_after, 5),
    ignoreItem: name => matches(ignore, name),
    scheduledItem: name => matches(scheduled, name),      // sold only on some days: absence is normal
    scheduledTabs: list(raw.scheduled_tabs),
    scheduledCategory: cat => !!cat && matches(schedCats, cat),
    mutedTabs: list(raw.muted_tabs),

    telegramDaily: /^(da|yes|1|true)$/i.test(String(raw.telegram_daily)),

// same product, different spelling: case, diacritics, HTML entities, punctuation, "330 ml" vs "330ml", "gr" vs "g"
function cosmeticKey(s) {
  return decode(s).normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ').replace(/(\d) (ml|l|g|gr|kg|cl|buc)\b/g, '$1$2').replace(/(\d)gr\b/g, '$1g').trim();
}

// cosmeticKey without quantities (330ml, 0.5l, 6 buc, 250g); empty if nothing but a quantity is left
function sizeFreeKey(s) {
  return cosmeticKey(s).replace(/\b\d+(ml|l|g|kg|cl|buc)\b/g, ' ').replace(/\s+/g, ' ').trim();
}

function ensureHeader(sh, headers) {
  if (String(sh.getRange(1, 1).getValue()) === headers[0]) return;
  sh.insertRowBefore(1);
  sh.getRange(1, 1, 1, headers.length).setValues([headers]);
  sh.setFrozenRows(1);
}

function dailyCheck() {
  const ss = SpreadsheetApp.getActive(), tz = ss.getSpreadsheetTimeZone(), F = CFG();
  SpreadsheetApp.flush();
  const fetchStatus = refreshSources();
  const today = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  const ymd = v => v instanceof Date ? Utilities.formatDate(v, tz, 'yyyy-MM-dd')
    : /^\d{4}-\d\d-\d\d/.test(String(v)) ? String(v).slice(0, 10)
    : v ? Utilities.formatDate(new Date(v), tz, 'yyyy-MM-dd') : today;
  const daysSince = d => Math.round((new Date(today) - new Date(d)) / 864e5);
  const round2 = x => Math.round(x * 100) / 100;
  const rep = ss.getSheetByName(REPORT_SHEET), snap = ss.getSheetByName(SNAP_SHEET), log = ss.getSheetByName(LOG_SHEET);
  ensureHeader(rep, REPORT_HEADERS);
  const idTabs = new Set(COMPETITORS.filter(c => c.idCol).map(c => c.sheet));
  const props = PropertiesService.getScriptProperties();

  // snapshot: Competitor | Key | Item | Price | Disc % | First seen | Missing
  // tabs without an ID column are keyed by cosmeticKey(name), so "OUĂ BENEDICT" and "Ouă Benedict" are the same item
  const snapRows = snap.getLastRow() > 1 ? snap.getRange(2, 1, snap.getLastRow() - 1, 8).getValues() : [];
  const old = {};
  snapRows.forEach(([c, key, name, price, disc, first, miss, cat]) => {
    const k = idTabs.has(c) ? String(key) : cosmeticKey(name);
    (old[c] = old[c] || {})[k] = { key: k, name: String(name), price, disc: Number(disc) || 0, first: ymd(first), miss: Number(miss) || 0, cat: String(cat || '') };
  });

  // recently delisted items, so a comeback is reported as RETURNED
  let remSh = ss.getSheetByName(REMOVED_SHEET);
  if (!remSh) { remSh = ss.insertSheet(REMOVED_SHEET); remSh.appendRow(['Competitor', 'Key', 'Item', 'Price', 'Removed on']); remSh.hideSheet(); }
  const gone = {};
  (remSh.getLastRow() > 1 ? remSh.getRange(2, 1, remSh.getLastRow() - 1, 5).getValues() : []).forEach(([c, key, name, price, since]) => {
    since = ymd(since);
    if (daysSince(since) <= F.forgetRemovedDays) (gone[c] = gone[c] || {})[String(key)] = { name: String(name), price, since };
  });

  const changes = [], logRows = [], newSnap = [];

  for (const cfg of COMPETITORS) {
    const sh = ss.getSheetByName(cfg.sheet);
    if (!sh) { logRows.push([today, cfg.sheet, 0, 'SHEET NOT FOUND']); continue; }
    const last = sh.getLastRow();
    if (last < 2) { Logger.log(`❌ ${cfg.sheet}: tab is empty`); continue; }
    if (oldLayoutTab_(sh, cfg)) {                                        // not converted yet (Bolt tabs wait for middayRefresh): do not read it with the new columns
      logRows.push([today, cfg.sheet, 0, 'FETCH FAILED (layout vechi – rulează migrateLayout)']);
      Object.values(old[cfg.sheet] || {}).forEach(r => newSnap.push([cfg.sheet, r.key, r.name, r.price, r.disc, r.first, r.miss, r.cat || '']));
      continue;
    }
    const col = c => last > 1 ? sh.getRange(`${c}2:${c}${last}`).getValues().flat() : [];
    const names = col(cfg.nameCol), prices = col(cfg.priceCol), ids = cfg.idCol ? col(cfg.idCol) : [], cats = cfg.catCol ? col(cfg.catCol) : [], promos = cfg.promoCol ? col(cfg.promoCol) : [];

    const origs = cfg.origCol ? col(cfg.origCol) : [];
    const cur = {};
    // Wolt / Bolt prices now include the required option (the can in a combo) and the Wolt campaign: one silent new baseline, no price alerts
    const rebase = /^(WOLT|BOLT)_MENU$/.test(cfg.fn || '') && !props.getProperty('layout2_' + cfg.sheet);
    const prev = rebase ? undefined : old[cfg.sheet], was = old[cfg.sheet] || {};
    names.forEach((n, k) => {
      const name = String(n || '').replace(/[ \s]+/g, ' ').trim(), shown = Number(prices[k]);
      if (!name || F.ignoreItem(name) || isNaN(shown) || shown <= 0) return;
      const o = Number(origs[k]);
      const list = (o > shown) ? o : shown;                       // real price
      const badge = promoPct(promos[k]);
      const disc = list > shown ? (badge > 0 ? badge : Math.round((1 - shown / list) * 100)) : 0;   // platform badge wins over our own rounding
      const key = cfg.idCol && ids[k] !== '' && ids[k] != null ? String(ids[k]) : cosmeticKey(name);
      cur[key] = { key, name, price: list, shown, disc, first: (was[key] && was[key].first) || today, miss: 0, cat: String(cats[k] || '').trim() };
    });
    const count = Object.keys(cur).length;
    const errCount = prices.filter(p => typeof p === 'string' && p.startsWith('#')).length;
    const fetchErr = fetchStatus[cfg.sheet];
    const scheduledTab = F.scheduledTabs.includes(cfg.sheet);           // menu legitimately shrinks outside its hours
    const failed = fetchErr || errCount > 0 || count === 0 || (!scheduledTab && prev && count < Object.keys(prev).length * 0.5);

    if (failed) {
      logRows.push([today, cfg.sheet, count, 'FETCH FAILED' + (fetchErr ? ` (${fetchErr})` : errCount ? ` (${errCount} error cells)` : count === 0 ? ' (0 produse valide)' : ` (${count} produse față de ${Object.keys(prev || {}).length} ieri)`)]);
      Object.values(was).forEach(r => newSnap.push([cfg.sheet, r.key, r.name, r.price, r.disc, r.first, r.miss, r.cat || '']));
      continue;
    }

    if (rebase) props.setProperty('layout2_' + cfg.sheet, '1');
    if (!prev) {
      logRows.push([today, cfg.sheet, count, rebase ? 'BASELINE (format nou)' : 'INITIAL BASELINE']);
    } else {
      let tabChanges = [];
      const push = (type, item, oldP, newP) => {
        const d = oldP !== '' && newP !== '' ? round2(newP - oldP) : 0;
        tabChanges.push([today, cfg.sheet, type, item, oldP, newP, d || '', d ? d / oldP : '']);
      };
      const pushDisc = (name, r, o) => {
        if (r.disc && r.disc !== o.disc) push('DISCOUNT', `${name} (-${r.disc}%)`, r.price, r.shown);
        else if (!r.disc && o.disc) push('DISCOUNT_ENDED', name, '', r.price);
      };
      const moved = (d, base) => d !== 0 && Math.abs(d) >= F.minDiffLei && Math.abs(d / base) >= F.minDiffPct;
      const added = [];

      for (const r of Object.values(cur)) {
        const o = prev[r.key];
        if (!o) { added.push(r); continue; }
        const d = round2(r.price - o.price);
        if (d !== 0 && !moved(d, o.price)) r.price = o.price;   // below threshold: keep old price so small steps add up
        const renamed = o.name !== r.name && cosmeticKey(o.name) !== cosmeticKey(r.name);
        if (renamed) push('RENAMED', `${o.name} → ${r.name}`, o.price, r.price);
        else if (moved(d, o.price)) push('PRICE_CHANGED', r.name, o.price, r.price);
        pushDisc(r.name, r, o);
      }
      const missing = Object.values(prev).filter(o => !cur[o.key]);   // includes items still inside the grace period

      // rename = a missing item and a new item that look alike: similar name, or one name contains the other at (almost) the same price
      const looksRenamed = (o, a) => {
        if (similarity(normName(o.name), normName(a.name)) >= RENAME_THRESHOLD) return true;
        // only the size changed ("Corona 355ml" → "Corona 330ml"): same name without quantities, price in the same ballpark
        const bo = sizeFreeKey(o.name);
        if (bo && bo === sizeFreeKey(a.name) && Math.abs(a.price - o.price) <= Math.max(2, o.price * 0.25)) return true;
        const x = ` ${cosmeticKey(o.name)} `, y = ` ${cosmeticKey(a.name)} `;
        return (x.includes(y) || y.includes(x)) && Math.abs(a.price - o.price) <= Math.max(1, o.price * 0.1);
      };
      const usedAdd = new Set();
      let twin;
      for (const o of missing) {
        if (F.scheduledItem(o.name) || scheduledTab || F.scheduledCategory(o.cat)) { newSnap.push([cfg.sheet, o.key, o.name, o.price, o.disc, o.first, 0, o.cat]); continue; }   // off today by schedule
        let best = null, bestScore = 0;
        for (const a of added) {
          if (usedAdd.has(a.key) || !looksRenamed(o, a)) continue;
          const s = similarity(normName(o.name), normName(a.name));
          if (s > bestScore) { bestScore = s; best = a; }
        }
        if (best) {
          usedAdd.add(best.key);
          if (cosmeticKey(o.name) !== cosmeticKey(best.name)) push('RENAMED', `${o.name} → ${best.name}`, o.price, best.price);
          else if (moved(round2(best.price - o.price), o.price)) push('PRICE_CHANGED', best.name, o.price, best.price);
          pushDisc(best.name, best, o);
          best.first = o.first;                                        // same product: keep its original first-seen date
          if (!best.cat) best.cat = o.cat;
        } else if (o.miss + 1 >= F.missingRuns && (twin = Object.values(cur).find(a => a.key !== o.key && a.first >= o.first && looksRenamed(o, a)))) {
          // about to be removed, but a newer product with a matching name and price exists → it was renamed days ago
          if (cosmeticKey(o.name) !== cosmeticKey(twin.name)) push('RENAMED', `${o.name} → ${twin.name}`, o.price, twin.price);
          twin.first = o.first;
        } else if (o.miss + 1 < F.missingRuns) {
          newSnap.push([cfg.sheet, o.key, o.name, o.price, o.disc, o.first, o.miss + 1, o.cat]);   // pending, keep quietly
        } else {
          push('REMOVED', `${o.name} (${daysSince(o.first)} zile)`, o.price, '');
          (gone[cfg.sheet] = gone[cfg.sheet] || {})[o.key] = { name: o.name, price: o.price, since: today };
        }
      }
      for (const a of added) {
        if (usedAdd.has(a.key)) continue;
        const g = gone[cfg.sheet] && gone[cfg.sheet][a.key];
        if (g) {
          push('RETURNED', `${a.name} (lipsă ~${daysSince(g.since) + F.missingRuns - 1} zile)`, g.price, a.shown);
          delete gone[cfg.sheet][a.key];
        } else push('ADDED', a.name + (a.disc ? ` (-${a.disc}%)` : ''), '', a.shown);
      }

      // platform reformatted its names (e.g. Wolt 18 Sep): one summary line instead of dozens of renames
      const plainRenames = tabChanges.filter(c => c[2] === 'RENAMED' && c[6] === '');
      if (plainRenames.length >= F.massRename.min && plainRenames.length >= count * F.massRename.share) {
        tabChanges = tabChanges.filter(c => !plainRenames.includes(c));
        tabChanges.push([today, cfg.sheet, 'RENAMED', `${plainRenames.length} produse redenumite deodată (schimbare de format pe platformă)`, '', '', '', '']);
      }
      changes.push(...tabChanges);
      logRows.push([today, cfg.sheet, count, 'OK']);
    }
    Object.values(cur).forEach(r => newSnap.push([cfg.sheet, r.key, r.name, r.price, r.disc, r.first, r.miss, r.cat || '']));
  }

  // stale detection
  const hist = log.getLastRow() > 1 ? log.getRange(2, 1, log.getLastRow() - 1, 4).getValues() : [];
  for (const cfg of COMPETITORS) {
    const todayRow = logRows.find(r => r[1] === cfg.sheet);
    if (!todayRow || !String(todayRow[3]).startsWith('FETCH')) continue;
    const past = hist.filter(r => r[1] === cfg.sheet).map(r => String(r[3])).reverse();
    let streak = 1;
    for (const s of past) { if (s.startsWith('FETCH')) streak++; else break; }
    if (streak === STALE_DAYS)
      changes.push([today, cfg.sheet, 'SOURCE_LOST', `No data for ${STALE_DAYS} days – closed on this platform?`, '', '', '', '']);
  }

  if (changes.length) rep.getRange(rep.getLastRow() + 1, 1, changes.length, 8).setValues(changes);
  log.getRange(log.getLastRow() + 1, 1, logRows.length, 4).setValues(logRows);
  snap.getRange(1, 1, 1, 8).setValues([['Competitor', 'Key', 'Item', 'Price', 'Disc %', 'First seen', 'Missing', 'Category']]);
  if (snap.getLastRow() > 1) snap.getRange(2, 1, snap.getLastRow() - 1, 8).clearContent();
  if (newSnap.length) snap.getRange(2, 1, newSnap.length, 8).setValues(newSnap);
  const hs = ss.getSheetByName(HISTORY_SHEET);
  if (hs && newSnap.length) {
    const histRows = newSnap.map(([c, key, name, price, disc]) => [today, c, name, price, disc || '']);
    hs.getRange(hs.getLastRow() + 1, 1, histRows.length, 5).setValues(histRows);
  }
  const remRows = [];
  Object.entries(gone).forEach(([c, m]) => Object.entries(m).forEach(([k, g]) => remRows.push([c, k, g.name, g.price, g.since])));
  if (remSh.getLastRow() > 1) remSh.getRange(2, 1, remSh.getLastRow() - 1, 5).clearContent();
  if (remRows.length) remSh.getRange(2, 1, remRows.length, 5).setValues(remRows);

  const anyFailed = logRows.some(r => String(r[3]).startsWith('FETCH'));
  if (changes.length || anyFailed) sendAlert(today, changes, logRows);
  try { buildListings(); } catch (e) { Logger.log('buildListings: ' + e.message); }
}


// One-time: rename competitor tabs and every reference to them (snapshot, reports, history, log, maps). Run once, then delete.
function renameCompetitorTabs() {
  const RENAMES = { 'Macros - G': 'Marcos - G', 'Macros - B': 'Marcos - B' };
  const ss = SpreadsheetApp.getActive();
  for (const [from, to] of Object.entries(RENAMES)) { const sh = ss.getSheetByName(from); if (sh && !ss.getSheetByName(to)) sh.setName(to); }
  // plain-value tabs: replace everywhere; map tabs hold formulas, so only their header row is touched
  const targets = [[SNAP_SHEET, 0], [REPORT_SHEET, 0], [HISTORY_SHEET, 0], [LOG_SHEET, 0], [REMOVED_SHEET, 0], ['Meta', 0], [COMPMAP_SHEET, 1], [MATRIX_SHEET, 1]];

    vals.forEach(row => row.forEach((v, i) => { if (RENAMES[v]) { row[i] = RENAMES[v]; hit++; } }));
    if (hit) { rng.setValues(vals); Logger.log(name + ': ' + hit + ' cells renamed'); }
  }
}



function buildMatrix() {
  const ss = SpreadsheetApp.getActive();
  const cm = ss.getSheetByName(COMPMAP_SHEET);
  const comps = cm.getRange(1, 2, 1, cm.getLastColumn() - 1).getValues()[0].map(String).filter(Boolean);
  const mx = ss.getSheetByName(MATRIX_SHEET) || ss.insertSheet(MATRIX_SHEET);
  mx.clearContents();
  const n = cm.getLastRow() - 1;                         // number of products
  const colL = i => { let s = ''; i++; while (i > 0) { const m = (i - 1) % 26; s = String.fromCharCode(65 + m) + s; i = (i - m - 1) / 26; } return s; };

  const header = ['Produs', 'Pret', 'Min piață', 'Medie piață', 'Noi vs medie', 'Cel mai ieftin'];
  comps.forEach(c => header.push(c, 'Pret'));
  mx.getRange(1, 1, 1, header.length).setValues([header]).setFontWeight('bold');

  const f = [];
  // our price: the site first, then the master tabs (name in A, reference price in B) for products that are not on the site
  const ourPrice = COMPMAP_OWN_SOURCES.slice().reverse()
    .reduce((inner, s) => `IFERROR(VLOOKUP($A2:$A; '${s.tab}'!A:${String.fromCharCode(64 + s.price)}; ${s.price}; FALSE); ${inner})`, '""');
  f.push(['=FILTER(CompMap!A2:A; CompMap!A2:A<>"")',
          `=ARRAYFORMULA(IF($A2:$A=""; ""; ${ourPrice}))`]);
  mx.getRange(2, 1, 1, 2).setFormulas(f);

  const rows = [];
  for (let r = 2; r <= n + 1; r++) {
    const row = [
      `=IFERROR(MIN(FILTER(G${r}:${r}; $G$1:$1="Pret"; ISNUMBER(G${r}:${r}))); "")`,
      `=IFERROR(AVERAGE(FILTER(G${r}:${r}; $G$1:$1="Pret"; ISNUMBER(G${r}:${r}))); "")`,
      `=IF(AND(B${r}<>""; D${r}<>""); B${r}/D${r}-1; "")`,
      `=IFERROR(INDEX($F$1:$1; MATCH(C${r}; G${r}:${r}; 0)); "")`,
    ];
    comps.forEach((c, k) => {
      const ic = colL(6 + 2 * k), pc = colL(7 + 2 * k);
      row.push(`=IF($A${r}=""; ""; IFERROR(VLOOKUP($A${r}; CompMap!$A:$ZZ; MATCH(${ic}$1; CompMap!$1:$1; 0); FALSE); ""))`);
      row.push(`=IF(${ic}${r}=""; ""; IFERROR(ARRAYFORMULA(INDEX(_Snapshot!$D:$D; MATCH(1; (_Snapshot!$A:$A=${ic}$1)*(_Snapshot!$C:$C=${ic}${r}); 0))); "?"))`);
    });
    rows.push(row);
  }
  if (rows.length) mx.getRange(2, 3, rows.length, rows[0].length).setFormulas(rows);
  mx.getRange(2, 5, n, 1).setNumberFormat('0.0%');
  mx.setFrozenRows(1); mx.setFrozenColumns(2);
}





function testAllSources() {
  const ss = SpreadsheetApp.getActive();
  SpreadsheetApp.flush();
  const seen = {};
  for (const cfg of COMPETITORS) {
    if (seen[cfg.sheet]) { Logger.log(`❌ ${cfg.sheet}: DUPLICATE entry in COMPETITORS`); continue; }
    seen[cfg.sheet] = true;
    const sh = ss.getSheetByName(cfg.sheet);
    if (!sh) { Logger.log(`❌ ${cfg.sheet}: sheet not found (check exact name)`); continue; }
    const last = sh.getLastRow();
    const names  = sh.getRange(`${cfg.nameCol}2:${cfg.nameCol}${last}`).getValues().flat();
    const prices = sh.getRange(`${cfg.priceCol}2:${cfg.priceCol}${last}`).getValues().flat();
    let ok = 0, badPrice = [], dupes = {}, blankName = 0;
    names.forEach((n, i) => {
      const nm = String(n || '').trim(), p = prices[i];
      if (!nm && (p === '' || p == null)) return;
      if (!nm) { blankName++; return; }
      if (typeof p !== 'number' || isNaN(p) || p <= 0) { badPrice.push(`${nm} → "${p}"`); return; }
      dupes[nm] = (dupes[nm] || 0) + 1; ok++;
    });
    const dupList = Object.entries(dupes).filter(([, c]) => c > 1).map(([n, c]) => `${n} ×${c}`);
    const flag = ok === 0 ? '❌' : (badPrice.length || dupList.length || blankName) ? '⚠️' : '✅';
    Logger.log(`${flag} ${cfg.sheet}: ${ok} valid items` +
      (badPrice.length ? ` | ${badPrice.length} bad prices: ${badPrice.slice(0, 3).join('; ')}` : '') +
      (blankName ? ` | ${blankName} rows with price but no name` : '') +
      (dupList.length ? ` | duplicate names: ${dupList.slice(0, 3).join('; ')}` : ''));
  }
}


function GLOVO_MENU(storeUrl) {
  storeUrl = String(storeUrl).trim();
  const html = UrlFetchApp.fetch(storeUrl, {
    muteHttpExceptions: true,
    headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/125 Safari/537.36' }
  }).getContentText();
  const num = t => parseFloat(t.replace(/[^\d,]/g, '').replace(',', '.'));
  const rows = [STD_HEADER];
  const re = /List_title__\w+"[^>]*>([^<]+)<\/h3>|ItemRow_titleContainer__\w+"><p[^>]*>([^<]+)<\/p>[\s\S]*?ItemRow_priceContainer__\w+">(.*?)<\/div>/g;
  let section = '', m;
  while ((m = re.exec(html)) !== null) {
    if (m[1]) { section = m[1]; continue; }
    const spans = [...m[3].matchAll(/<span[^>]*>([^<]+)<\/span>/g)].map(x => x[1]);
    const price = num(spans[0] || ''), orig = spans[1] ? num(spans[1]) : '';
    const promo = (m[0].match(/ItemRow_tagsContainer[\s\S]*?(-\d+%)/) || [])[1] || '';   // Glovo's own badge, e.g. -10% (its prices round differently)
    rows.push(stdRow(decode(m[2]), price, orig, promo ? parseInt(promo.replace(/[^\d]/g, ''), 10) : '', decode(section)));   // plain number (30): a '-30%' text becomes -0.3 in the sheet
  }
  LAST_META = glovoMeta(html);
  return rows.length > 1 ? rows : 'NO ITEMS – check URL or blocked';
}


// Variant naming shared by the site readers (same as burgerbrasov.ro): "X – Single" -> "X", "X – Meniu" -> "Meniu X", any other option -> "X – option"
function variantName_(name, opt) {
  opt = String(opt || "").trim();
  if (!opt || /^single$/i.test(opt)) return name;
  return /^meniu$/i.test(opt) && !/^meniu/i.test(name) ? `Meniu ${name}` : `${name} – ${opt}`;
}

function WOO_MENU(siteUrl) {
  const base = siteUrl.replace(/\/+$/, '');
  const hdr = { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0' } };
  const minor = p => Math.pow(10, p.prices.currency_minor_unit);
  const fetchAll = q => {
    const out = [];
    for (let page = 1; page <= 10; page++) {
      const res = UrlFetchApp.fetch(`${base}/wp-json/wc/store/v1/products?per_page=100&page=${page}&${q}`, hdr);
      if (res.getResponseCode() !== 200) return page === 1 ? null : out;
      const items = JSON.parse(res.getContentText());
      out.push(...items);
      if (items.length < 100) break;
    }
    return out;
  };
  const products = fetchAll('type=simple') , variations = fetchAll('type=variation');
  if (!products) return 'HTTP error';
  const rows = [STD_HEADER];
  // WooCommerce sends names HTML-encoded ("Steak &#038; Eggs", "The &#8217;Merican"): decode() turns them into plain text
  const wooRow = (name, x, cats, id) => {
    const price = Number(x.prices.price) / minor(x), reg = Number(x.prices.regular_price) / minor(x);
    return stdRow(name, price, reg, reg > price ? Math.round((1 - price / reg) * 100) : 0, cats.map(c => decode(c.name)).join(', '), '', id);
  };
  products.forEach(p => rows.push(wooRow(decode(p.name), p, p.categories, p.id)));
  (variations || []).forEach(v => {
    const opt = decode((v.variation || '').replace(/^[^:]+:\s*/, ''));       // "Extra: Cu parmezan" -> "Cu parmezan"
    rows.push(wooRow(variantName_(decode(v.name), opt), v, v.categories || [], v.id));
  });
  return rows;
}

// promo badge as stored in a tab: 30, '-30%', or -0.3 (when the sheet turned '-30%' into a percentage cell) → 30
function promoPct(v) {
  if (v === '' || v == null) return 0;
  if (typeof v === 'number') return Math.round(Math.abs(v) <= 1 ? Math.abs(v) * 100 : Math.abs(v));
  const n = parseInt(String(v).replace(/[^\d]/g, ''), 10);
  return n > 0 ? n : 0;
}

function decode(t) {
  return String(t).replace(/&#(\d+);/g, (_, n) => String.fromCharCode(n)).replace(/&#x([0-9a-f]+);/gi, (_, h) => String.fromCharCode(parseInt(h, 16)))
    .replace(/&amp;/g, '&').replace(/&quot;/g, '"').replace(/&#039;|&apos;/g, "'").replace(/&nbsp;/g, ' ').trim();
}

function refreshTabs_(pick) {
  const ss = SpreadsheetApp.getActive();
  for (const cfg of COMPETITORS.filter(pick)) {
    const sh = ss.getSheetByName(cfg.sheet); if (!sh) continue;
    let data;
    for (let attempt = 1; attempt <= 3; attempt++) {
      try { data = this[cfg.fn](cfg.url); } catch (e) { data = 'ERR ' + e.message; }
      if (Array.isArray(data)) break;
      Utilities.sleep(8000 * attempt);
    }
    Utilities.sleep(5000);                       // Bolt rate-limits bursts
    const oldCount = Math.max(sh.getLastRow() - 1, 0);
    if (!Array.isArray(data) || data.length < 2 || (oldCount && data.length - 1 < oldCount * 0.5)) {
      Logger.log(`${cfg.sheet}: kept previous data (${String(data).slice(0, 60)}${Array.isArray(data) ? ', ' + (data.length - 1) + ' items' : ''})`);
      continue;                                   // closed / partial menu → keep yesterday's full list
    }
    writeDataTab_(sh, data);
    writeMeta(ss, cfg.sheet);
  }
}

// Ratings / fees only, for every Glovo, Wolt and Bolt tab — no menu changes, no daily email. Run by hand any time.
function refreshMeta() {
  const ss = SpreadsheetApp.getActive();
  for (const cfg of COMPETITORS) {
    if (!cfg.url || !/^(GLOVO|WOLT|BOLT)_MENU$/.test(cfg.fn)) continue;
    LAST_META = null;
    try {
      if (cfg.fn === 'GLOVO_MENU') {
        const html = UrlFetchApp.fetch(cfg.url, { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/125 Safari/537.36' } }).getContentText();
        LAST_META = glovoMeta(html);
      } else if (cfg.fn === 'WOLT_MENU') {
        LAST_META = woltMeta(cfg.url);
      } else {
        const m = String(cfg.url).match(/\/p\/(\d+)/);
        if (m) { LAST_META = boltMeta(m[1]); Utilities.sleep(4000); }   // Bolt rate-limits bursts
      }
    } catch (e) { Logger.log(`${cfg.sheet}: ${e.message}`); }
    Logger.log(`${cfg.sheet}: ${LAST_META ? JSON.stringify(LAST_META) : 'no data'}`);
    writeMeta(ss, cfg.sheet);
  }
}

// Meta tab: one row per platform tab per run — rating as shown by the platform (Bolt /5, Wolt /10, Glovo %),
// reviews (Bolt, Wolt), delivery fee, ETA (Bolt) and a comparable "Rating %" column.
function writeMeta(ss, tab) {
  if (!LAST_META) return;
  const ms = ss.getSheetByName('Meta') || ss.insertSheet('Meta');
  const headers = ['Date', 'Competitor', 'Rating', 'Reviews', 'Delivery fee', 'Delivery min', 'Rating %'];
  if (ms.getLastRow() === 0) ms.appendRow(headers);
  else if (ms.getLastColumn() < headers.length) ms.getRange(1, 1, 1, headers.length).setValues([headers]);
  ms.appendRow([Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), 'yyyy-MM-dd'), tab,
    LAST_META.rating, LAST_META.reviews, LAST_META.fee, LAST_META.minutes, LAST_META.pct]);
  LAST_META = null;
}
