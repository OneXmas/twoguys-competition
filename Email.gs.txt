// ===== EMAIL CONFIG — edit here =====
const EMAIL_CFG = {
  subject: (nComp, nOwn, date) => `🍔 Competiție: ${nComp ? `${nComp} ${plural(nComp)}` : 'fără modificări'}${nOwn ? ` · la noi: ${nOwn}` : ''} (${date})`,
  title: 'Raport competiție',
  footer: 'Raport automat generat zilnic numai dacă există modificări față de ziua precedentă',
  ownTitle: 'Canalele noastre',
  labels: {
    PRICE_CHANGED: 'Preț NOU', DISCOUNT: 'PROMO', DISCOUNT_ENDED: 'Promo End', ADDED: 'ADĂUGAT',
    RETURNED: 'REVENIT', REMOVED: 'DELISTAT', RENAMED: 'REDENUMIT', SOURCE_LOST: 'Sursă pierdută',
  },
  order: ['SOURCE_LOST', 'PRICE_CHANGED', 'DISCOUNT', 'DISCOUNT_ENDED', 'ADDED', 'RETURNED', 'REMOVED', 'RENAMED'],
  colors: {
    accent:   '#ff6b00',   // top/bottom bar + header
    ADDED:    '#16a34a',   // green
    RETURNED: '#65a30d',   // olive
    REMOVED:  '#6b7280',   // grey
    RENAMED:  '#0891b2',   // teal
    DISCOUNT: '#db2777',   // pink
    DISCOUNT_ENDED: '#9ca3af',
    up:       '#dc2626',   // price went up — red
    down:     '#2563eb',   // price went down — blue
    headerBg: '#111827',
    rowAlt:   '#f9fafb',
    SOURCE_LOST: '#7c3aed',   // purple
  },
};

const plural = n => n === 1 ? 'modificare' : 'modificări';
const isOwnTab = tab => OWN_CHANNELS.includes(tab) || tab === 'Burgerbrasov';
const PLATFORM_NAMES = { G: 'Glovo', W: 'Wolt', B: 'Bolt' };
const displayName = tab => String(tab).replace(/ - ([GWB])$/, (_, p) => ` - ${PLATFORM_NAMES[p]}`);   // "Dafud - G" → "Dafud - Glovo"


function sendAlert(today, changes, logRows) {
  const C = EMAIL_CFG, col = C.colors, F = CFG();
  changes = changes.filter(r => !F.mutedTabs.includes(r[1]));          // muted tabs: tracked in the sheet, left out of the email
  if (!changes.length && !logRows.some(r => String(r[3]).startsWith('FETCH'))) return;
  const esc = s => String(decode(s)).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  const fmt = v => v === '' || v == null ? '' : Number(v).toFixed(2);
  const badge = (type, color) => `<span style="display:inline-block;padding:2px 8px;border-radius:10px;background:${color};color:#fff;font-size:11px;font-weight:700;">${C.labels[type] || type}</span>`;
  const rank = t => { const i = C.order.indexOf(t); return i < 0 ? 99 : i; };

  const byComp = {};
  changes.forEach(r => (byComp[r[1]] = byComp[r[1]] || []).push(r));

  const shopBlock = (comp, rows) => {
    rows = [...rows].sort((a, b) => rank(a[2]) - rank(b[2]));
    // long ADDED/REMOVED/RETURNED lists (menu reshuffle, category switched off) → one row each
    const out = [];
    for (const type of [...new Set(rows.map(r => r[2]))]) {
      const same = rows.filter(r => r[2] === type);
      if (['ADDED', 'REMOVED', 'RETURNED'].includes(type) && same.length > F.collapseAfter) {
        const shown = same.slice(0, 8).map(r => r[3]).join(', ');
        out.push([today, comp, type, `${same.length} produse: ${shown}${same.length > 8 ? `, … (+${same.length - 8})` : ''}`, '', '', '', '']);
      } else out.push(...same);
    }
    let html = `
    <div style="font-size:15px;font-weight:700;color:#111827;margin:18px 0 6px;text-align:left;">${esc(displayName(comp))}
      <span style="font-weight:400;color:#6b7280;font-size:13px;">— ${rows.length} ${plural(rows.length)}</span></div>
    <table width="100%" cellpadding="6" cellspacing="0" border="0" style="border:1px solid #e5e7eb;border-radius:8px;font-size:13px;">
      <tr style="background:${col.headerBg};color:#fff;">
        <th align="center">Modificare</th><th align="center">Articol</th><th align="center">Vechi</th><th align="center">Nou</th><th align="center">Dif</th>
      </tr>`;
    out.forEach((r, i) => {
      const type = r[2], d = r[6];
      const badgeColor = ['PRICE_CHANGED', 'RETURNED'].includes(type) && d !== '' ? (d > 0 ? col.up : col.down) : (col[type] || '#6b7280');
      const diffTxt = d === '' || d == null || type === 'DISCOUNT' ? '' : `${d > 0 ? '▲' : '▼'} ${fmt(Math.abs(d))} (${(Math.abs(r[7]) * 100).toFixed(1)}%)`;   // promo: the -% is already in the item name
      html += `
      <tr style="background:${i % 2 ? col.rowAlt : '#fff'};">
        <td>${badge(type, badgeColor)}</td>
        <td align="left">${esc(r[3])}</td>
        <td align="right">${fmt(r[4])}</td>
        <td align="right"><b>${fmt(r[5])}</b></td>
        <td align="right" style="color:${badgeColor};font-weight:700;">${diffTxt}</td>
      </tr>`;
    });
    return html + '</table>';
  };

  const byPrio = (a, b) => (F.priorityTabs.includes(b) - F.priorityTabs.includes(a)) || a.localeCompare(b);
  const comps = Object.keys(byComp).filter(c => !isOwnTab(c)).sort(byPrio), own = Object.keys(byComp).filter(isOwnTab);
  const count = list => list.reduce((s, c) => s + byComp[c].length, 0);
  const nComp = count(comps), nOwn = count(own);

  // one-line summary: how many of each change type among competitors
  const perType = {};
  comps.forEach(c => byComp[c].forEach(r => perType[r[2]] = (perType[r[2]] || 0) + 1));
  const summary = Object.keys(perType).sort((a, b) => rank(a) - rank(b))
    .map(t => `${badge(t, col[t] || (t === 'PRICE_CHANGED' ? col.up : '#6b7280'))} ${perType[t]}`).join(' &nbsp; ');

  let body = summary ? `<div style="margin:10px 0 4px;font-size:13px;">${summary}</div>` : '';
  comps.forEach(c => body += shopBlock(c, byComp[c]));
  if (!comps.length) body += `<p style="font-size:13px;color:#6b7280;">Nicio modificare la concurență azi.</p>`;
  if (own.length) {
    body += `<div style="font-size:16px;font-weight:800;color:#111827;margin:28px 0 0;padding-top:14px;border-top:2px solid ${col.accent};text-align:left;">${C.ownTitle}</div>`;
    own.forEach(c => body += shopBlock(c, byComp[c]));
  }

  const failed = logRows.filter(r => String(r[3]).startsWith('FETCH'));
  const warn = failed.length
    ? `<div style="margin-top:16px;padding:10px;background:#fef3c7;border-left:4px solid #f59e0b;font-size:13px;text-align:left;">
        ⚠️ Neverificat azi (eroare la citire): <b>${esc(failed.map(r => displayName(r[1])).join(', '))}</b></div>` : '';


  const html = `
  <div style="background:#f3f4f6;padding:20px 8px;font-family:Arial,sans-serif;">
  <div style="max-width:640px;margin:0 auto;background:#fff;border-radius:12px;overflow:hidden;border:1px solid #e5e7eb;">
    <div style="height:5px;background:${col.accent};"></div>
    <div style="padding:20px;border-bottom:1px solid #e5e7eb;text-align:center;">
      <div style="font-size:18px;font-weight:800;color:#111827;">${C.title}</div>
      <div style="font-size:13px;color:#6b7280;">${today}</div>
      ${body}
      ${warn}
      <div style="font-size:11px;color:#9ca3af;margin-top:16px;text-align:center;">${C.footer}</div>
    </div>
    <div style="height:5px;background:${col.accent};"></div>
  </div></div>`;

  if (F.telegramDaily && comps.length) {
    const lines = [`🍔 <b>Competiție: ${nComp} ${plural(nComp)}</b> (${today})`];
    comps.forEach(c => {
      const rows = byComp[c].sort((a, b) => rank(a[2]) - rank(b[2]));
      lines.push(`\n<b>${tgEsc(displayName(c))}</b>`);
      rows.slice(0, 6).forEach(r => lines.push(`• ${C.labels[r[2]] || r[2]}: ${tgEsc(r[3])}${r[4] !== '' && r[5] !== '' ? ` ${fmt(r[4])} → ${fmt(r[5])}` : r[5] !== '' ? ` ${fmt(r[5])}` : ''}`));
      if (rows.length > 6) lines.push(`  … +${rows.length - 6}`);
    });
    tgSend(lines.join('\n'));
  }
  MailApp.sendEmail({ to: F.alertEmail, subject: changes.length ? C.subject(nComp, nOwn, today) : `⚠️ Sursă fără date (${today})`, htmlBody: html });
}


function weeklyDigest() {
  const ss = SpreadsheetApp.getActive(), tz = ss.getSpreadsheetTimeZone();
  const today = new Date(), since = new Date(today.getTime() - 7 * 864e5);
  const fmt = d => Utilities.formatDate(d, tz, 'yyyy-MM-dd');
  const sinceStr = fmt(since), todayStr = fmt(today);
  const C = EMAIL_CFG, col = C.colors;
  const esc = s => String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;');
  const table = (headers, rows, aligns = []) => rows.length ? `
    <table width="100%" cellpadding="6" cellspacing="0" style="border:1px solid #e5e7eb;font-size:13px;margin-bottom:14px;">
      <tr style="background:${col.headerBg};color:#fff;">${headers.map((h, i) => `<th align="${aligns[i] || 'left'}">${h}</th>`).join('')}</tr>
      ${rows.map((r, i) => `<tr style="background:${i % 2 ? col.rowAlt : '#fff'};">${r.map((c, j) => `<td align="${aligns[j] || 'left'}">${c}</td>`).join('')}</tr>`).join('')}
    </table>` : '<p style="color:#6b7280;font-size:13px;">—</p>';
  const section = (title, html) => `<div style="font-size:15px;font-weight:700;margin:20px 0 8px;">${title}</div>${html}`;

  // 1. changes last 7 days, per competitor and type
  const rep = ss.getSheetByName(REPORT_SHEET);
  const rows = rep.getLastRow() > 1 ? rep.getRange(2, 1, rep.getLastRow() - 1, 8).getValues() : [];
  const week = rows.filter(r => { const d = r[0] instanceof Date ? fmt(r[0]) : String(r[0]); return d >= sinceStr; });
  const types = ['PRICE_CHANGED', 'DISCOUNT', 'ADDED', 'REMOVED', 'RENAMED'];
  const perComp = {};
  week.forEach(r => { const c = perComp[r[1]] = perComp[r[1]] || {}; c[r[2]] = (c[r[2]] || 0) + 1; });
  const summaryRows = Object.entries(perComp).sort((a, b) => b[1].length - a[1].length)
    .map(([c, t]) => [esc(c), ...types.map(k => t[k] || '')]);

  // 2. biggest price moves (list price only)
  const movers = week.filter(r => r[2] === 'PRICE_CHANGED' && r[7] !== '')
    .sort((a, b) => Math.abs(b[7]) - Math.abs(a[7])).slice(0, 8)
    .map(r => [esc(r[1]), esc(r[3]), Number(r[4]).toFixed(2), Number(r[5]).toFixed(2),
      `<span style="color:${r[6] > 0 ? col.up : col.down};font-weight:700;">${r[6] > 0 ? '▲' : '▼'} ${(Math.abs(r[7]) * 100).toFixed(1)}%</span>`]);

  // 3. active discounts right now (from snapshot)
  const snap = ss.getSheetByName(SNAP_SHEET);
  const snapRows = snap.getLastRow() > 1 ? snap.getRange(2, 1, snap.getLastRow() - 1, 5).getValues() : [];
  const promos = snapRows.filter(r => Number(r[4]) > 0).sort((a, b) => b[4] - a[4]).slice(0, 15)
    .map(r => [esc(r[0]), esc(r[2]), Number(r[3]).toFixed(2), `-${r[4]}%`]);

  // 4. our position vs market (MatchMatrix)
  let position = '';
  const mx = ss.getSheetByName(MATRIX_SHEET);
  if (mx && mx.getLastRow() > 1) {
    const hdr = mx.getRange(1, 1, 1, mx.getLastColumn()).getValues()[0];
    const iItem = 0, iAD = hdr.indexOf('Noi vs medie'), iAE = hdr.indexOf('Cel mai ieftin'), iAC = hdr.indexOf('Medie piață');
    const data = mx.getRange(2, 1, mx.getLastRow() - 1, mx.getLastColumn()).getValues()
      .filter(r => r[iItem] && typeof r[iAD] === 'number');
    if (data.length) {
      const avg = data.reduce((s, r) => s + r[iAD], 0) / data.length;
      const fmtPct = v => `<span style="color:${v > 0 ? col.up : col.down};font-weight:700;">${v > 0 ? '+' : ''}${(v * 100).toFixed(1)}%</span>`;
      const sorted = [...data].sort((a, b) => b[iAD] - a[iAD]);
      const posRows = [...sorted.slice(0, 5), ...sorted.slice(-5)].filter((r, i, a) => a.indexOf(r) === i)
        .map(r => [esc(r[iItem]), Number(r[1]).toFixed(2), Number(r[iAC]).toFixed(2), fmtPct(r[iAD]), esc(r[iAE] || '')]);
      position = `<p style="font-size:14px;">Poziție medie față de piață: ${fmtPct(avg)} pe ${data.length} produse</p>` +
        table(['Produs', 'Preț nostru', 'Medie piață', 'Diferență', 'Cel mai ieftin'], posRows, ['left', 'right', 'right', 'right', 'left']);
    }
  }

  // 5. data health
  const log = ss.getSheetByName(LOG_SHEET);
  const logRows = log.getLastRow() > 1 ? log.getRange(2, 1, log.getLastRow() - 1, 4).getValues() : [];
  const failed = {};
  logRows.filter(r => (r[0] instanceof Date ? fmt(r[0]) : String(r[0])) >= sinceStr && String(r[3]).startsWith('FETCH'))
    .forEach(r => failed[r[1]] = (failed[r[1]] || 0) + 1);
  let health = Object.keys(failed).length
    ? `<div style="padding:10px;background:#fef3c7;border-left:4px solid #f59e0b;font-size:13px;">⚠️ Zile fără date: ${Object.entries(failed).map(([c, n]) => `${esc(c)} (${n})`).join(', ')}</div>`
    : `<p style="font-size:13px;color:${col.ADDED};">✅ Toate sursele au livrat date în fiecare zi.</p>`;
  const cmS = ss.getSheetByName(COMPMAP_SHEET);
  if (cmS && cmS.getLastRow() > 1) {
    const v = cmS.getRange(2, 2, cmS.getLastRow() - 1, cmS.getLastColumn() - 1).getValues().flat().map(String);
    const blank = v.filter(x => !x).length, unrev = v.filter(x => /^[~≈]/.test(x)).length;
    health += `<p style="font-size:13px;">CompMap: ${blank} celule nepotrivite, ${unrev} sugestii nerevizuite (~/≈)</p>`;
  }

  // 6. own channels vs site
  let channels = '';
  const ch = ss.getSheetByName('Channels');
  if (ch && ch.getLastRow() > 1) {
    const hdr = ch.getRange(1, 1, 1, ch.getLastColumn()).getValues()[0].map(String);
    const data = ch.getRange(2, 1, ch.getLastRow() - 1, ch.getLastColumn()).getValues().filter(r => r[0]);
    const chRows = [];
    hdr.forEach((h, i) => {
      if (h !== 'Δ%') return;
      const name = hdr[i - 2], diffs = data.map(r => r[i]).filter(v => typeof v === 'number');
      const missing = data.filter(r => !r[i - 2]).length;
      const avg = diffs.length ? diffs.reduce((s, v) => s + v, 0) / diffs.length : null;
      const worst = data.filter(r => typeof r[i] === 'number').sort((a, b) => b[i] - a[i])[0];
      chRows.push([esc(name), avg === null ? '—' : `<b>${avg > 0 ? '+' : ''}${(avg * 100).toFixed(1)}%</b>`, diffs.length, missing,
        worst ? `${esc(worst[0])} (${worst[i] > 0 ? '+' : ''}${(worst[i] * 100).toFixed(0)}%)` : '—']);
    });
    channels = table(['Canal', 'Markup mediu', 'Produse comparate', 'Lipsă în Map', 'Cel mai mare markup'], chRows, ['left','right','right','right','left']);
  }

  const html = `
  <div style="background:#f3f4f6;padding:20px 8px;font-family:Arial,sans-serif;">
  <div style="max-width:680px;margin:0 auto;background:#fff;border-radius:12px;overflow:hidden;border:1px solid #e5e7eb;">
    <div style="height:5px;background:${col.accent};"></div>
    <div style="padding:20px;border-bottom:1px solid #e5e7eb;text-align:center;">
      <div style="font-size:18px;font-weight:800;">Raport săptămânal concurență</div>
      <div style="font-size:13px;color:#6b7280;">${sinceStr} → ${todayStr} · ${week.length} ${plural(week.length)}</div>
    </div>
    <div style="padding:4px 20px 20px;">
      ${section('Poziția noastră', position || '<p style="color:#6b7280;font-size:13px;">MatchMatrix gol</p>')}
      ${section('Canale proprii vs Restaurant Passage', channels || '<p style="color:#6b7280;font-size:13px;">Foaia Channels lipsește</p>')}
      ${section('Modificări pe concurent', table(['Concurent', 'Preț', 'Discount', 'Nou', 'Eliminat', 'Redenumit'], summaryRows, ['left','right','right','right','right','right']))}
      ${section('Cele mai mari mișcări de preț', table(['Concurent', 'Produs', 'Vechi', 'Nou', 'Δ'], movers, ['left','left','right','right','right']))}
      ${section('Promoții active acum', table(['Concurent', 'Produs', 'Preț listă', 'Discount'], promos, ['left','left','right','right']))}
      ${section('Sănătatea datelor', health)}
    </div>
    <div style="height:5px;background:${col.accent};"></div>
  </div></div>`;

  MailApp.sendEmail({ to: CFG().weeklyEmail, subject: `📊 Raport săptămânal concurență (${sinceStr} → ${todayStr})`, htmlBody: html });
}




// Weekly platform standing: for every Glovo / Wolt / Bolt tab, the latest rating, reviews and delivery fee from the Meta tab,
// compared with the values from 7 days earlier. Ranked per platform by Rating %, own shops in bold.
function weeklyMetaReport() {
  const ss = SpreadsheetApp.getActive(), tz = ss.getSpreadsheetTimeZone();
  const ms = ss.getSheetByName('Meta');
  if (!ms || ms.getLastRow() < 2) { Logger.log('Meta tab is empty'); return; }
  const fmt = d => Utilities.formatDate(d, tz, 'yyyy-MM-dd');
  const today = new Date(), todayStr = fmt(today), weekAgo = fmt(new Date(today.getTime() - 7 * 864e5));
  const ymd = v => v instanceof Date ? fmt(v) : String(v).slice(0, 10);
  const rows = ms.getRange(2, 1, ms.getLastRow() - 1, 7).getValues().filter(r => r[1]);
  const num = v => v === '' || v == null ? null : Number(String(v).replace('%', '').replace(',', '.'));

  // per tab: latest row, and the latest row dated on/before a week ago
  const byTab = {};
  rows.forEach(r => {
    const d = ymd(r[0]), t = byTab[r[1]] = byTab[r[1]] || { now: null, before: null };
    if (!t.now || d >= ymd(t.now[0])) t.now = r;              // same day twice (manual refreshMeta): the later row wins
    if (d <= weekAgo && (!t.before || d > ymd(t.before[0]))) t.before = r;
  });

  const C = EMAIL_CFG, col = C.colors;
  const esc = s => String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  const delta = (a, b, digits, suffix) => {
    if (a == null || b == null) return '<span style="color:#9ca3af;">—</span>';
    const d = a - b;
    if (Math.abs(d) < 0.005) return '<span style="color:#9ca3af;">=</span>';
    return `<span style="color:${d > 0 ? col.ADDED : col.up};font-weight:700;">${d > 0 ? '▲' : '▼'} ${Math.abs(d).toFixed(digits)}${suffix || ''}</span>`;
  };
  const newReviews = (a, b) => {
    if (a == null) return '<span style="color:#9ca3af;">—</span>';
    if (b == null) return '<span style="color:#9ca3af;">—</span>';
    const d = a - b;
    return d > 0 ? `<b style="color:${col.ADDED};">+${d}</b>` : '<span style="color:#9ca3af;">0</span>';
  };
  const own = new Set(OWN_CHANNELS), skip = new Set(CFG().scheduledTabs);   // breakfast venues: not rated yet
  // Glovo rating: the sheet may have turned "100%" into 1 (a percentage cell) → always work in whole percent
  const ratingOf = (suffix, v) => { const r = num(v); return r == null ? null : suffix === 'G' && r <= 1 ? Math.round(r * 100) : r; };
  const pctOf = (suffix, rating) => rating == null ? null : suffix === 'B' ? Math.round(rating / 5 * 100) : suffix === 'W' ? Math.round(rating * 10) : rating;
  const showRating = (suffix, r) => r == null ? '—' : suffix === 'G' ? r + '%' : r.toFixed(suffix === 'B' ? 2 : 1);
  // Wolt only publishes review counts in steps (20, 50, 100, 200, 500…), so show "50+" and report "Noi" only when the step goes up
  const showReviews = (suffix, v) => v == null ? '—' : suffix === 'W' ? v + '+' : String(v);
  const sections = [];
  for (const [suffix, label] of [['G', 'Glovo'], ['W', 'Wolt'], ['B', 'Bolt']]) {
    const list = Object.entries(byTab).filter(([tab]) => tab.endsWith(' - ' + suffix) && !skip.has(tab))
      .map(([tab, t]) => { const r = ratingOf(suffix, t.now[2]); return { tab, now: t.now, before: t.before, rating: r, pct: pctOf(suffix, r) }; })
      .sort((a, b) => ((b.pct == null ? -1 : b.pct) - (a.pct == null ? -1 : a.pct)) || ((b.rating || 0) - (a.rating || 0)));
    if (!list.length) continue;
    const trs = list.map((x, i) => {
      const n = x.now, b = x.before || [];
      const name = esc(x.tab.replace(/ - [GWB]$/, ''));
      const stale = ymd(n[0]) < weekAgo ? ` <span style="color:#d97706;font-size:11px;" title="ultima citire">${ymd(n[0])}</span>` : '';
      const before = ratingOf(suffix, b[2]), reviews = num(n[3]), reviewsBefore = num(b[3]);
      const newRev = suffix === 'W' && reviews != null && reviewsBefore != null
        ? (reviews > reviewsBefore ? `<b style="color:${col.ADDED};">↑ ${reviews}+</b>` : '<span style="color:#9ca3af;">—</span>')
        : newReviews(reviews, reviewsBefore);
      return `<tr style="background:${i % 2 ? col.rowAlt : '#fff'};${own.has(x.tab) ? 'font-weight:700;' : ''}">
        <td align="center">${i + 1}</td><td>${name}${stale}</td>
        <td align="right">${showRating(suffix, x.rating)}</td><td align="right">${delta(x.rating, before, suffix === 'G' ? 0 : suffix === 'B' ? 2 : 1, suffix === 'G' ? '%' : '')}</td>
        <td align="right">${showReviews(suffix, reviews)}</td><td align="right">${newRev}</td></tr>`;
    }).join('');
    sections.push(`<div style="font-size:15px;font-weight:700;margin:20px 0 8px;">${label}</div>
      <table width="100%" cellpadding="6" cellspacing="0" style="border:1px solid #e5e7eb;font-size:13px;">
      <tr style="background:${col.headerBg};color:#fff;"><th>#</th><th align="left">Magazin</th><th align="right">Rating</th><th align="right">Δ 7 zile</th>
      <th align="right">Recenzii</th><th align="right">Noi</th></tr>${trs}</table>`);
  }

  const html = `
