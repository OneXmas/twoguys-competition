function mapNorm(s) {
  return normName(s)
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')          // ș → s, ă → a
    .replace(/\b\d+(g|gr|ml|l|kg|buc)\b/g, ' ')               // 280g, 500ml
    .replace(/\bthe\b|\bburger\b|\bsingle\b|\bsimplu\b/g, ' ')
    .replace(/['’`]/g, '')
    .replace(/\s+/g, ' ').trim();
}

function normName(s) {
  return String(s || '').toLowerCase()
    .replace(/[\u00a0\s]+/g, ' ').replace(/[’‘]/g, "'").replace(/[“”]/g, '"')
    .replace(/\b(si|și|and|cu|de|la)\b|&/g, ' ')
    .replace(/(\d)\s+(g|gr|ml|l|kg)\b/g, '$1$2')
    .replace(/[^\p{L}\p{N}]+/gu, ' ').trim();
}

function similarity(a, b) {          // Dice coefficient on character bigrams
  const grams = s => { const m = {}; for (let i = 0; i < s.length - 1; i++) { const g = s.slice(i, i + 2); m[g] = (m[g] || 0) + 1; } return m; };
  const A = grams(a), B = grams(b); let hit = 0;
  for (const g in A) if (B[g]) hit += Math.min(A[g], B[g]);
  const tot = (a.length - 1) + (b.length - 1);
  return tot > 0 ? (2 * hit) / tot : 0;
}

function priceOf(ss, tab) {
  const cfg = COMPETITORS.find(c => c.sheet === tab), sh = ss.getSheetByName(tab), m = {};
  if (!cfg || !sh || sh.getLastRow() < 2) return m;
  const n = sh.getRange(`${cfg.nameCol}2:${cfg.nameCol}${sh.getLastRow()}`).getValues().flat();
  const p = sh.getRange(`${cfg.priceCol}2:${cfg.priceCol}${sh.getLastRow()}`).getValues().flat();
  n.forEach((x, i) => { if (x) m[String(x).trim()] = Number(p[i]); });
  return m;
}

function prefillSheet(sheetName, marker, onlyNew) {       // onlyNew: fuzzy-fill just the columns added by this call
  const ss = SpreadsheetApp.getActive();
  const map = ss.getSheetByName(sheetName);
  let headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String).filter(Boolean);
  const hadColumn = new Set(headers);

  // CompMap: make sure every competitor has a column
  if (sheetName === COMPMAP_SHEET) {
    COMPETITORS.map(c => c.sheet).filter(t => !OWN_CHANNELS.includes(t) && t.toLowerCase() !== 'burgerbrasov' && !headers.includes(t))
      .forEach(t => {
        headers.push(t);
        if (map.getMaxColumns() < headers.length) map.insertColumnsAfter(map.getMaxColumns(), headers.length - map.getMaxColumns());
        map.getRange(1, headers.length).setValue(t);
      });
  }

  const prices = headers.map(t => priceOf(ss, t));
  const namesOf = tab => Object.keys(prices[headers.indexOf(tab)]);
  // CompMap: column A = all our products (site + masters); Map (legacy): the tab named in its first header
  const ownList = sheetName === COMPMAP_SHEET ? ownProducts() : null;
  const ownPrice = {}; if (ownList) ownList.forEach(p => ownPrice[cosmeticKey(p.name)] = p.price);
  const site = ownList || namesOf(headers[0]).map(name => ({ name }));
  const existing = map.getLastRow() > 1 ? map.getRange(2, 1, map.getLastRow() - 1, headers.length).getValues() : [];
  const rows = existing.map(r => r.map(String));
  const known = new Set(rows.map(r => cosmeticKey(r[0]))), knownKey = new Set(rows.map(r => ownKey_(r[0])));
  site.forEach(p => {
    const cos = cosmeticKey(p.name), key = ownKey_(p.name);
    if (known.has(cos) || (ownList && !p.noMerge && knownKey.has(key))) return;
    known.add(cos); knownKey.add(key);
    rows.push([p.name, ...headers.slice(1).map(() => '')]);
  });

  headers.slice(1).forEach((tab, ci) => {
    if (onlyNew && hadColumn.has(tab)) return;
    const c = ci + 1, pool = namesOf(tab);
    const used = new Set(rows.map(r => r[c].replace(/^~/, '')).filter(Boolean));
    for (const r of rows) {
      if (r[c]) continue;
      let best = '', bestScore = 0;
      const a = mapNorm(r[0]), base = ownList ? ownPrice[cosmeticKey(r[0])] : prices[0][r[0]];
      for (const cand of pool) {
        if (used.has(cand)) continue;
        const p = prices[c][cand];
        if (base && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) continue;
        const s = similarity(a, mapNorm(cand));
        if (s > bestScore) { bestScore = s; best = cand; }
      }
      if (bestScore >= MAP_THRESHOLD) { r[c] = marker + best; used.add(best); }
    }
  });

  rows.sort((x, y) => x[0].localeCompare(y[0]));
  map.getRange(2, 1, Math.max(map.getLastRow() - 1, 1), headers.length).clearContent();
  map.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

// Where our products come from, in priority order (the first source that has a product gives its price). Column A is always the
// name; price = column number of the price; type/typeIs = only rows whose column `type` equals `typeIs` count.
// Add { tab: 'Breakfast', price: 2 } to benchmark breakfast items too.
const COMPMAP_OWN_SOURCES = [
  { tab: 'Burgerbrasov',     price: 3 },                                                  // restaurant site (standard layout): A name, C price
  { tab: 'MASTER Twoguys',   price: 3, type: 15, typeIs: 'Produs din lista' },            // Mypassage list: A name, C price, O row type
  { tab: 'TwoGuys Home',     price: 2 },                                                  // master tabs: A name, B reference price
  { tab: 'TwoGuys Delivery', price: 2 },
  { tab: 'Smash Delivery',   price: 2, noMerge: true },                                  // a different brand: never folded into a TwoGuys product
];

// same product, different spelling across our shops ("Two Guys" on the site, "Two Guys Burger" on Mypassage): same name without
// filler words and the same quantities ("330ml") = same product
function ownKey_(name) { return mapNorm(name) + '|' + qtyOf(name).join(','); }

// Every product of ours worth comparing with the market: [{ name, price, src }], each product once (spelling variants merged into
// the first source's name). out.merged lists the variants that were folded in.
function ownProducts() {
  const ss = SpreadsheetApp.getActive(), F = CFG(), out = [], seenCos = new Set(), seenKey = new Set();
  out.merged = [];
  const add = (name, price, src, noMerge) => {
    name = String(name || '').replace(/[\s ]+/g, ' ').trim();
    const cos = cosmeticKey(name);
    if (!name || !cos || F.ignoreItem(name) || seenCos.has(cos)) return;
    seenCos.add(cos);
    const key = ownKey_(name);
    if (!noMerge && seenKey.has(key)) { out.merged.push(`${name} (${src})`); return; }
    seenKey.add(key);
    out.push({ name, price: Number(price) > 0 ? Number(price) : '', src, noMerge: !!noMerge });
  };
  COMPMAP_OWN_SOURCES.forEach(s => {
    const sh = ss.getSheetByName(s.tab);
    if (!sh || sh.getLastRow() < 2) return;
    const width = Math.min(sh.getLastColumn(), Math.max(2, s.price, s.type || 0));
    sh.getRange(2, 1, sh.getLastRow() - 1, width).getValues().forEach(r => {
      if (s.type && r[s.type - 1] !== s.typeIs) return;
      add(r[0], r[s.price - 1], s.tab, s.noMerge);
    });
  });
  return out;
}

// Adds to CompMap column A every product of ours that is not there yet (competitor cells empty). Never deletes or edits a name;
// rows stay sorted A→Z. Safe to re-run. Run it after a new product appears on the site or on a master tab.
function syncCompMapProducts() {
  const ss = SpreadsheetApp.getActive(), cm = ss.getSheetByName(COMPMAP_SHEET);
  const width = cm.getLastColumn();
  const existing = cm.getLastRow() > 1 ? cm.getRange(2, 1, cm.getLastRow() - 1, width).getValues().map(r => r.map(String)).filter(r => r.some(Boolean)) : [];
  const haveCos = new Set(existing.map(r => cosmeticKey(r[0])).filter(Boolean)), haveKey = new Set(existing.map(r => ownKey_(r[0])));
  const own = ownProducts();
  const isVariant = p => !p.noMerge && haveKey.has(ownKey_(p.name));          // same product as an existing row, other spelling
  const fresh = own.filter(p => !haveCos.has(cosmeticKey(p.name)) && !isVariant(p));
  const variants = own.filter(p => !haveCos.has(cosmeticKey(p.name)) && isVariant(p)).map(p => `${p.name} (${p.src})`).concat(own.merged);
  const bySrc = {}; fresh.forEach(p => bySrc[p.src] = (bySrc[p.src] || 0) + 1);
  if (variants.length) Logger.log(`Not added, same product as an existing row under another spelling (${variants.length}): ${variants.slice(0, 20).join('; ')}`);
  const rows = existing.concat(fresh.map(p => [p.name, ...Array(width - 1).fill('')])).sort((a, b) => a[0].localeCompare(b[0]));
  if (cm.getMaxRows() < rows.length + 1) cm.insertRowsAfter(cm.getMaxRows(), rows.length + 1 - cm.getMaxRows());
  cm.getRange(1, 1).setValue('Produs');                       // was "BurgerBrasov": the column now holds all our shops
  cm.getRange(2, 1, Math.max(cm.getLastRow() - 1, 1), width).clearContent();
  if (rows.length) cm.getRange(2, 1, rows.length, width).setValues(rows);
  Logger.log(`CompMap: ${fresh.length} products added (${Object.entries(bySrc).map(([s, n]) => s + ' ' + n).join(', ') || 'none'}), ${rows.length} in total`);
}

function clearSuspiciousMap() {
  const ss = SpreadsheetApp.getActive(), map = ss.getSheetByName(MAP_SHEET);
  const headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String);
  const prices = headers.map(t => priceOf(ss, t));
  const rng = map.getRange(2, 1, map.getLastRow() - 1, headers.length), rows = rng.getValues();
  let cleared = 0;
  rows.forEach(r => {
    const base = prices[0][String(r[0]).trim()];
    if (!base) return;
    for (let c = 1; c < headers.length; c++) {
      const p = prices[c][String(r[c]).trim()];
      if (r[c] && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) { r[c] = ''; cleared++; }
    }
  });
  rng.setValues(rows);
  Logger.log(`Cleared ${cleared} suspicious mappings`);
}

function llmFillCompMap(onlyTabs) {                      // onlyTabs: optional list of competitor tabs; default = every column
  const ss = SpreadsheetApp.getActive();
  const key = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_KEY');
  if (!key) throw new Error('Set ANTHROPIC_KEY in Script properties');
  const map = ss.getSheetByName(COMPMAP_SHEET);
  const headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String).filter(Boolean);
  const rng = map.getRange(2, 1, map.getLastRow() - 1, headers.length), rows = rng.getValues().map(r => r.map(String));
  const ours = {}; ownProducts().forEach(p => ours[cosmeticKey(p.name)] = p.price);      // our price per product, whichever shop it is from

  // The AI answers about BATCH of our items at a time (a long list would not fit in one answer). "No equivalent" is written as "-"
  // so the cell is not asked again; a cell left blank (bad answer) is retried on the next run. Apps Script stops after 6 min:
  // when the time is up the rest is left for the next run, which continues where this one stopped.
  const BATCH = 80, t0 = Date.now(), skipped = [];
  headers.slice(1).forEach((tab, ci) => {
    if (Array.isArray(onlyTabs) && !onlyTabs.includes(tab)) return;
    const c = ci + 1, theirs = priceOf(ss, tab), tried = new Set();
    while (true) {
    if (Date.now() - t0 > 270000) { skipped.push(tab); break; }
    const blanks = rows.filter(r => r[0] && !r[c] && !tried.has(r)).slice(0, BATCH);
    if (!blanks.length) break;
    blanks.forEach(r => tried.add(r));
    const taken = new Set(rows.map(r => r[c].replace(/^[~≈]/, '')).filter(Boolean));
    const candidates = Object.entries(theirs).filter(([n]) => !taken.has(n));
    if (!candidates.length) break;

    const prompt = `You match restaurant menu items between two Romanian burger restaurants for price comparison.
Match by PRODUCT EQUIVALENCE (same category and role on the menu), not identical recipe:
- a chicken burger matches their chicken burger even if ingredients differ
- "Meniu X" (burger + fries + drink) only matches their menu/combo version; a plain burger only matches a plain burger
- ribs match ribs ("coaste"), fries match fries, a specific drink matches the same drink
- return null when there is no reasonable equivalent. Never force a match.
Each competitor item may be used at most once.

OUR items (name | price RON):
${blanks.map(r => `${r[0]} | ${ours[cosmeticKey(r[0])] || '?'}`).join('\n')}

COMPETITOR "${tab}" items (name | price RON):
${candidates.map(([n, p]) => `${n} | ${p}`).join('\n')}

Respond with ONLY a JSON object, no prose, no markdown: {"<our item>": "<competitor item exactly as written>" or null, ...}. Include every one of OUR items as a key.`;

    let res;
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
          method: 'post', contentType: 'application/json', muteHttpExceptions: true,
          headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' },
          payload: JSON.stringify({ model: LLM_MODEL, max_tokens: 6000, messages: [{ role: 'user', content: prompt }] })
        });
        if (res.getResponseCode() < 500) break;
      } catch (e) { if (attempt === 3) { Logger.log(`${tab}: network error ${e.message}`); res = null; } }
      Utilities.sleep(2000 * attempt);
    }
    if (!res) break;
    const body = JSON.parse(res.getContentText());
    if (res.getResponseCode() !== 200 || !body.content) { Logger.log(`${tab}: API ${res.getResponseCode()} ${res.getContentText().slice(0, 300)}`); break; }
    const text = body.content.map(b => b.text || '').join('');
    let out; try { out = JSON.parse(text.replace(/```json|```/g, '').trim()); } catch (e) { Logger.log(`${tab}: bad JSON: ${text.slice(0, 200)}`); continue; }

    const valid = new Set(candidates.map(([n]) => n)), used = new Set();
    let filled = 0, none = 0;
    blanks.forEach(r => {
      if (!Object.prototype.hasOwnProperty.call(out, r[0])) return;   // not answered: stays blank, retried next run
      const v = out[r[0]];
      if (v === null) { r[c] = '-'; none++; return; }                 // the AI found no equivalent
      if (!valid.has(v) || used.has(v)) return;                       // only real names, no reuse
      r[c] = LLM_MARKER + v; used.add(v); filled++;
    });
    Logger.log(`${tab}: ${filled} suggestions, ${none} no equivalent, for ${blanks.length} blanks`);
    }
  });
  rng.setValues(rows);
  if (skipped.length) Logger.log('Time limit reached, not done yet (run llmFillCompMap again): ' + skipped.join(', '));
}


function migrateCompMap() {
  const OWN_CHANNELS = ['Mypassage', 'TwoGuys - G', 'TwoGuys - W'];
  const ss = SpreadsheetApp.getActive();
  const mx = ss.getSheetByName(MATRIX_SHEET);
  const hdr = mx.getRange(1, 1, 1, mx.getLastColumn()).getValues()[0].map(String);
  const data = mx.getRange(2, 1, mx.getLastRow() - 1, mx.getLastColumn()).getValues();

  // competitor item columns = header is a tab name in COMPETITORS (price columns have blank headers)
  const tabs = COMPETITORS.map(c => c.sheet).filter(t => !OWN_CHANNELS.includes(t) && t.toLowerCase() !== 'burgerbrasov');
  const compHeaders = tabs;                                       // every competitor tab, in COMPETITORS order
  const itemCols = compHeaders.map(h => hdr.indexOf(h));          // -1 if not in the matrix yet

  // existing pairs keyed by our product name
  const pairs = {};
  data.forEach(r => {
    const prod = String(r[0]).trim(); if (!prod) return;
    pairs[prod] = pairs[prod] || {};
    itemCols.forEach((ci, k) => { if (ci < 0) return; const v = String(r[ci]).trim(); if (v) pairs[prod][compHeaders[k]] = v; });
  });

  // all site products (so CompMap is complete, not just rows that had matches)
  const site = ss.getSheetByName('BurgerBrasov');
  site.getRange(2, 1, site.getLastRow() - 1, 1).getValues().flat().map(v => String(v).trim()).filter(Boolean)
    .forEach(p => pairs[p] = pairs[p] || {});

  let cm = ss.getSheetByName(COMPMAP_SHEET) || ss.insertSheet(COMPMAP_SHEET);
  cm.clearContents();
  const rows = Object.keys(pairs).sort((a, b) => a.localeCompare(b))
    .map(p => [p, ...compHeaders.map(h => pairs[p][h] || '')]);
  cm.getRange(1, 1, 1, compHeaders.length + 1).setValues([['BurgerBrasov', ...compHeaders]]);
  if (rows.length) cm.getRange(2, 1, rows.length, compHeaders.length + 1).setValues(rows);
  cm.setFrozenRows(1); cm.setFrozenColumns(1);
  Logger.log(`CompMap: ${rows.length} products, ${compHeaders.length} competitors: ${compHeaders.join(', ')}`);
}


const LLM_MODEL = 'claude-sonnet-5-5';
const LLM_MARKER = '≈';
