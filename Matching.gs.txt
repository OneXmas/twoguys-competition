function mapNorm(s) {
  return normName(s)
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')          // ș → s, ă → a
    .replace(/\b\d+(g|gr|ml|l|kg|buc)\b/g, ' ')               // 280g, 500ml
    .replace(/\bthe\b|\bburger\b|\bsingle\b|\bsimplu\b/g, ' ')
    .replace(/['’`]/g, '')
    .replace(/\s+/g, ' ').trim();
}

function normName(s) {
  return String(s || '').toLowerCase()
    .replace(/[\u00a0\s]+/g, ' ').replace(/[’‘]/g, "'").replace(/[“”]/g, '"')
    .replace(/\b(si|și|and|cu|de|la)\b|&/g, ' ')
    .replace(/(\d)\s+(g|gr|ml|l|kg)\b/g, '$1$2')
    .replace(/[^\p{L}\p{N}]+/gu, ' ').trim();
}

function similarity(a, b) {          // Dice coefficient on character bigrams
  const grams = s => { const m = {}; for (let i = 0; i < s.length - 1; i++) { const g = s.slice(i, i + 2); m[g] = (m[g] || 0) + 1; } return m; };
  const A = grams(a), B = grams(b); let hit = 0;
  for (const g in A) if (B[g]) hit += Math.min(A[g], B[g]);
  const tot = (a.length - 1) + (b.length - 1);
  return tot > 0 ? (2 * hit) / tot : 0;
}

function priceOf(ss, tab) {
  const cfg = COMPETITORS.find(c => c.sheet === tab), sh = ss.getSheetByName(tab), m = {};
  if (!cfg || !sh || sh.getLastRow() < 2) return m;
  const n = sh.getRange(`${cfg.nameCol}2:${cfg.nameCol}${sh.getLastRow()}`).getValues().flat();
  const p = sh.getRange(`${cfg.priceCol}2:${cfg.priceCol}${sh.getLastRow()}`).getValues().flat();
  n.forEach((x, i) => { if (x) m[String(x).trim()] = Number(p[i]); });
  return m;
}

function prefillSheet(sheetName, marker) {
  const ss = SpreadsheetApp.getActive();
  const map = ss.getSheetByName(sheetName);
  let headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String).filter(Boolean);

  // CompMap: make sure every competitor has a column
  if (sheetName === COMPMAP_SHEET) {
    COMPETITORS.map(c => c.sheet).filter(t => !OWN_CHANNELS.includes(t) && t.toLowerCase() !== 'burgerbrasov' && !headers.includes(t))
      .forEach(t => { headers.push(t); map.getRange(1, headers.length).setValue(t); });
  }

  const prices = headers.map(t => priceOf(ss, t));
  const namesOf = tab => Object.keys(prices[headers.indexOf(tab)]);
  const site = namesOf(headers[0]);
  const existing = map.getLastRow() > 1 ? map.getRange(2, 1, map.getLastRow() - 1, headers.length).getValues() : [];
  const rows = existing.map(r => r.map(String));
  const known = new Set(rows.map(r => r[0]));
  site.forEach(n => { if (!known.has(n)) rows.push([n, ...headers.slice(1).map(() => '')]); });

  headers.slice(1).forEach((tab, ci) => {
    const c = ci + 1, pool = namesOf(tab);
    const used = new Set(rows.map(r => r[c].replace(/^~/, '')).filter(Boolean));
    for (const r of rows) {
      if (r[c]) continue;
      let best = '', bestScore = 0;
      const a = mapNorm(r[0]), base = prices[0][r[0]];
      for (const cand of pool) {
        if (used.has(cand)) continue;
        const p = prices[c][cand];
        if (base && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) continue;
        const s = similarity(a, mapNorm(cand));
        if (s > bestScore) { bestScore = s; best = cand; }
      }
      if (bestScore >= MAP_THRESHOLD) { r[c] = marker + best; used.add(best); }
    }
  });

  rows.sort((x, y) => x[0].localeCompare(y[0]));
  map.getRange(2, 1, Math.max(map.getLastRow() - 1, 1), headers.length).clearContent();
  map.getRange(2, 1, rows.length, headers.length).setValues(rows);
}

function prefillMap()     { prefillSheet(MAP_SHEET, ''); }
function prefillCompMap() { prefillSheet(COMPMAP_SHEET, '~'); }

function clearSuspiciousMap() {
  const ss = SpreadsheetApp.getActive(), map = ss.getSheetByName(MAP_SHEET);
  const headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String);
  const prices = headers.map(t => priceOf(ss, t));
  const rng = map.getRange(2, 1, map.getLastRow() - 1, headers.length), rows = rng.getValues();
  let cleared = 0;
  rows.forEach(r => {
    const base = prices[0][String(r[0]).trim()];
    if (!base) return;
    for (let c = 1; c < headers.length; c++) {
      const p = prices[c][String(r[c]).trim()];
      if (r[c] && p && (p / base < MAP_PRICE_BAND[0] || p / base > MAP_PRICE_BAND[1])) { r[c] = ''; cleared++; }
    }
  });
  rng.setValues(rows);
  Logger.log(`Cleared ${cleared} suspicious mappings`);
}

function llmFillCompMap() {
  const ss = SpreadsheetApp.getActive();
  const key = PropertiesService.getScriptProperties().getProperty('ANTHROPIC_KEY');
  if (!key) throw new Error('Set ANTHROPIC_KEY in Script properties');
  const map = ss.getSheetByName(COMPMAP_SHEET);
  const headers = map.getRange(1, 1, 1, map.getLastColumn()).getValues()[0].map(String).filter(Boolean);
  const rng = map.getRange(2, 1, map.getLastRow() - 1, headers.length), rows = rng.getValues().map(r => r.map(String));
  const ours = priceOf(ss, headers[0]);

  headers.slice(1).forEach((tab, ci) => {
    const c = ci + 1;
    const blanks = rows.filter(r => r[0] && !r[c]);
    if (!blanks.length) return;
    const theirs = priceOf(ss, tab);
    const taken = new Set(rows.map(r => r[c].replace(/^[~≈]/, '')).filter(Boolean));
    const candidates = Object.entries(theirs).filter(([n]) => !taken.has(n));
    if (!candidates.length) return;

    const prompt = `You match restaurant menu items between two Romanian burger restaurants for price comparison.
Match by PRODUCT EQUIVALENCE (same category and role on the menu), not identical recipe:
- a chicken burger matches their chicken burger even if ingredients differ
- "Meniu X" (burger + fries + drink) only matches their menu/combo version; a plain burger only matches a plain burger
- ribs match ribs ("coaste"), fries match fries, a specific drink matches the same drink
- return null when there is no reasonable equivalent. Never force a match.
Each competitor item may be used at most once.

OUR items (name | price RON):
${blanks.map(r => `${r[0]} | ${ours[r[0]] || '?'}`).join('\n')}

COMPETITOR "${tab}" items (name | price RON):
${candidates.map(([n, p]) => `${n} | ${p}`).join('\n')}

Respond with ONLY a JSON object, no prose, no markdown: {"<our item>": "<competitor item exactly as written>" or null, ...}. Include every one of OUR items as a key.`;

    let res;
    for (let attempt = 1; attempt <= 3; attempt++) {
      try {
        res = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', {
          method: 'post', contentType: 'application/json', muteHttpExceptions: true,
          headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' },
          payload: JSON.stringify({ model: LLM_MODEL, max_tokens: 4000, messages: [{ role: 'user', content: prompt }] })
        });
        if (res.getResponseCode() < 500) break;
      } catch (e) { if (attempt === 3) { Logger.log(`${tab}: network error ${e.message}`); return; } }
      Utilities.sleep(2000 * attempt);
    }
    const body = JSON.parse(res.getContentText());
    if (res.getResponseCode() !== 200 || !body.content) { Logger.log(`${tab}: API ${res.getResponseCode()} ${res.getContentText().slice(0, 300)}`); return; }
    const text = body.content.map(b => b.text || '').join('');
    let out; try { out = JSON.parse(text.replace(/```json|```/g, '').trim()); } catch (e) { Logger.log(`${tab}: bad JSON: ${text.slice(0, 200)}`); return; }

    const valid = new Set(candidates.map(([n]) => n)), used = new Set();
    let filled = 0;
    rows.forEach(r => {
      const v = out[r[0]];
      if (r[c] || !v || !valid.has(v) || used.has(v)) return;   // only blanks, only real names, no reuse
      r[c] = LLM_MARKER + v; used.add(v); filled++;
    });
    Logger.log(`${tab}: ${filled} suggestions for ${blanks.length} blanks`);
  });
  rng.setValues(rows);
}


function migrateCompMap() {
  const OWN_CHANNELS = ['Mypassage', 'TwoGuys - G', 'TwoGuys - W'];
  const ss = SpreadsheetApp.getActive();
  const mx = ss.getSheetByName(MATRIX_SHEET);
  const hdr = mx.getRange(1, 1, 1, mx.getLastColumn()).getValues()[0].map(String);
  const data = mx.getRange(2, 1, mx.getLastRow() - 1, mx.getLastColumn()).getValues();

  // competitor item columns = header is a tab name in COMPETITORS (price columns have blank headers)
  const tabs = COMPETITORS.map(c => c.sheet).filter(t => !OWN_CHANNELS.includes(t) && t.toLowerCase() !== 'burgerbrasov');
  const compHeaders = tabs;                                       // every competitor tab, in COMPETITORS order
  const itemCols = compHeaders.map(h => hdr.indexOf(h));          // -1 if not in the matrix yet

  // existing pairs keyed by our product name
  const pairs = {};
  data.forEach(r => {
    const prod = String(r[0]).trim(); if (!prod) return;
    pairs[prod] = pairs[prod] || {};
    itemCols.forEach((ci, k) => { if (ci < 0) return; const v = String(r[ci]).trim(); if (v) pairs[prod][compHeaders[k]] = v; });
  });

  // all site products (so CompMap is complete, not just rows that had matches)
  const site = ss.getSheetByName('BurgerBrasov');
  site.getRange(2, 1, site.getLastRow() - 1, 1).getValues().flat().map(v => String(v).trim()).filter(Boolean)
    .forEach(p => pairs[p] = pairs[p] || {});

  let cm = ss.getSheetByName(COMPMAP_SHEET) || ss.insertSheet(COMPMAP_SHEET);
  cm.clearContents();
  const rows = Object.keys(pairs).sort((a, b) => a.localeCompare(b))
    .map(p => [p, ...compHeaders.map(h => pairs[p][h] || '')]);
  cm.getRange(1, 1, 1, compHeaders.length + 1).setValues([['BurgerBrasov', ...compHeaders]]);
  if (rows.length) cm.getRange(2, 1, rows.length, compHeaders.length + 1).setValues(rows);
  cm.setFrozenRows(1); cm.setFrozenColumns(1);
  Logger.log(`CompMap: ${rows.length} products, ${compHeaders.length} competitors: ${compHeaders.join(', ')}`);
}
